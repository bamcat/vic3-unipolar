smd_leftturn.2 = {
	type = country_event
	placement = ROOT
	title = smd_leftturn.2.t
	desc = smd_leftturn.2.d
	flavor = smd_leftturn.2.f
	event_image = { video = "middleeast_oil_derricks" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		has_journal_entry = smd_je_left_turn
		any_scope_building = {
			OR = {
				is_building_type = building_oil_rig
				is_building_type = building_gas_field
			}
			country_ownership_fraction >= 0.5
		}
		market_capital.market ?= {
			mg:oil ?= { market_goods_pricier > 0.25 }
		}
	}

	option = {
		name = smd_leftturn.2.a
		default_option = yes
		add_treasury = {
			value = gdp
			multiply = 0.01
		}
		je:smd_je_left_turn ?= {
			add_progress = { value = 5 name = smd_leftturn_bar }
		}
		ai_chance = { base = 100 }
	}
}
