namespace = smd_language

smd_language.1 = {
	type = country_event
	placement = ROOT
	title = smd_language.1.t
	desc = smd_language.1.d
	flavor = smd_language.1.f
	event_image = { video = "ip2_cross_cultural_festival" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
	}

	immediate = {
		var:smd_language_minority ?= { save_scope_as = smd_language_minority }
	}

	option = {
		name = smd_language.1.a
		default_option = yes
		set_variable = smd_language_assimilation
		add_modifier = {
			name = smd_language_assimilation_policy
			days = long_modifier_time
		}
		if = {
			limit = { exists = scope:smd_language_minority }
			add_radicals = {
				value = medium_radicals
				culture = scope:smd_language_minority
			}
		}
		ai_chance = { base = 40 }
	}

	option = {
		name = smd_language.1.b
		set_variable = smd_language_rights
		add_modifier = {
			name = smd_language_rights_policy
			days = long_modifier_time
		}
		if = {
			limit = { exists = scope:smd_language_minority }
			add_loyalists = {
				value = small_radicals
				culture = scope:smd_language_minority
			}
		}
		ai_chance = { base = 60 }
	}
}
