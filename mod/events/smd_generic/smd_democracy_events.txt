smd_democracy.3 = {
	type = country_event
	placement = ROOT
	title = smd_democracy.3.t
	desc = smd_democracy.3.d
	flavor = smd_democracy.3.f
	event_image = { video = "middleeast_courtroom_upheaval" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		has_journal_entry = smd_je_democratic_consolidation
		has_variable = smd_recent_law_enacted
		has_variable = smd_dc_struck_law
		has_variable = smd_dc_prev_law_type
		has_law = var:smd_dc_struck_law
		institution_investment_level = {
			institution = institution_courts
			value >= 1
		}
	}

	immediate = {
		var:smd_dc_struck_law = {
			save_scope_as = smd_dc_struck_law_scope
		}
	}

	option = {
		name = smd_democracy.3.a
		default_option = yes
		activate_law = var:smd_dc_prev_law_type
		smd_dc_clear_struck_law = yes
		add_modifier = {
			name = smd_court_respected
			days = short_modifier_time
		}
		je:smd_je_democratic_consolidation ?= {
			add_progress = { value = 6 name = smd_democracy_bar }
		}
		ai_chance = { base = 70 }
	}

	option = {
		name = smd_democracy.3.b
		smd_dc_clear_struck_law = yes
		add_modifier = {
			name = smd_court_defied
			days = short_modifier_time
		}
		je:smd_je_democratic_consolidation ?= {
			add_progress = { value = -12 name = smd_democracy_bar }
		}
		ai_chance = { base = 30 }
	}
}

smd_democracy.4 = {
	type = country_event
	placement = ROOT
	title = smd_democracy.4.t
	desc = smd_democracy.4.d
	flavor = smd_democracy.4.f
	event_image = { video = "unspecific_military_parade" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		has_journal_entry = smd_je_democratic_consolidation
		ig:ig_armed_forces ?= { ig_approval <= 0 }
	}

	option = {
		name = smd_democracy.4.a
		trigger = {
			military_wage_level < high
		}
		set_military_wage_level = high
		je:smd_je_democratic_consolidation ?= {
			add_progress = { value = 3 name = smd_democracy_bar }
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = smd_democracy.4.b
		default_option = yes
		add_modifier = {
			name = smd_generals_retired
			days = short_modifier_time
		}
		ig:ig_armed_forces ?= {
			if = {
				limit = { is_in_government = yes }
				remove_ruling_interest_group = yes
			}
		}
		je:smd_je_democratic_consolidation ?= {
			add_progress = { value = 5 name = smd_democracy_bar }
		}
		ai_chance = { base = 40 }
	}
}
