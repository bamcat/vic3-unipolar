namespace = smd_contagion

smd_contagion.1 = {
	type = country_event
	placement = ROOT
	title = smd_contagion.1.t
	desc = smd_contagion.1.d
	flavor = smd_contagion.1.f
	event_image = { video = "votp_barricade" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		any_political_movement = { has_variable = smd_contagion_target }
	}

	immediate = {
		smd_contagion_save_movement = yes
	}

	option = {
		name = smd_contagion.1.a
		default_option = yes
		scope:smd_contagion_movement ?= {
			add_modifier = {
				name = smd_contagion_movement_watched
				days = 730
			}
		}
		je:smd_je_fear_of_contagion ?= {
			add_progress = { value = -10 name = smd_contagion_bar }
		}
		ai_chance = { base = 50 }
	}

	option = {
		name = smd_contagion.1.b
		scope:smd_contagion_movement ?= {
			add_modifier = {
				name = smd_contagion_movement_tolerated
				days = 730
			}
		}
		add_modifier = {
			name = smd_alarm_dismissed
			days = short_modifier_time
		}
		je:smd_je_fear_of_contagion ?= {
			add_progress = { value = 5 name = smd_contagion_bar }
		}
		ai_chance = { base = 50 }
	}
}

smd_contagion.3 = {
	type = country_event
	placement = ROOT
	title = smd_contagion.3.t
	desc = smd_contagion.3.d
	flavor = smd_contagion.3.f
	event_image = { video = "ip3_police_raid" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	duration = 3

	trigger = {
		any_political_movement = { has_variable = smd_contagion_target }
	}

	immediate = {
		smd_contagion_save_movement = yes
	}

	option = {
		name = smd_contagion.3.a
		default_option = yes
		scope:smd_contagion_movement ?= {
			add_modifier = {
				name = smd_contagion_movement_crushed
				days = normal_modifier_time
			}
		}
		add_modifier = {
			name = smd_emergency_powers
			days = normal_modifier_time
		}
		if = {
			limit = {
				enacting_any_law = no
				NOT = { has_law_or_variant = law_type:law_secret_police }
				any_law = {
					type = law_type:law_secret_police
					can_be_enacted = yes
				}
			}
			start_enactment = law_type:law_secret_police
			add_enactment_phase = 1
		}
		smd_contagion_clear_movement = yes
		ai_chance = { base = 60 }
	}

	option = {
		name = smd_contagion.3.b
		scope:smd_contagion_movement ?= {
			add_modifier = {
				name = smd_contagion_movement_emboldened
				days = short_modifier_time
			}
		}
		add_modifier = {
			name = smd_emergency_refused
			days = short_modifier_time
		}
		smd_contagion_clear_movement = yes
		ai_chance = { base = 40 }
	}
}
