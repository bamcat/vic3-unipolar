namespace = smd_refugee

smd_refugee.1 = {
	type = country_event
	placement = ROOT
	title = smd_refugee.1.t
	desc = smd_refugee.1.d
	flavor = smd_refugee.1.f
	event_image = { video = "southamerica_war_civilians" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
	}

	immediate = {
		var:smd_refugee_source ?= { save_scope_as = smd_refugee_source }
	}

	option = {
		name = smd_refugee.1.a
		default_option = yes
		add_modifier = {
			name = smd_border_opened
			days = 1825
		}
		if = {
			limit = { exists = scope:smd_refugee_source }
			scope:smd_refugee_source = {
				random_primary_culture = { save_scope_as = smd_refugee_culture }
			}
			capital = {
				create_mass_migration = {
					origin = scope:smd_refugee_source
					culture = scope:smd_refugee_culture
				}
			}
		}
		je:smd_je_refugee_crisis ?= {
			add_progress = { value = 10 name = smd_refugee_bar }
		}
		ai_chance = { base = 50 }
	}

	option = {
		name = smd_refugee.1.b
		add_modifier = {
			name = smd_border_closed
			days = 1825
		}
		je:smd_je_refugee_crisis ?= {
			add_progress = { value = -10 name = smd_refugee_bar }
		}
		ai_chance = { base = 50 }
	}
}
