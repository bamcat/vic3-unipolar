smd_devolution.2 = {
	type = country_event
	placement = ROOT
	title = smd_devolution.2.t
	desc = smd_devolution.2.d
	flavor = smd_devolution.2.f
	event_image = { video = "unspecific_politicians_arguing" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		has_journal_entry = smd_je_devolution
		exists = var:smd_devolution_state
	}

	immediate = {
		var:smd_devolution_state ?= { save_scope_as = smd_devolution_state }
	}

	option = {
		name = smd_devolution.2.a
		default_option = yes
		add_treasury = {
			value = gdp
			multiply = -0.002
		}
		var:smd_devolution_state ?= {
			add_modifier = {
				name = smd_regional_compromise
				days = 730
			}
		}
		je:smd_je_devolution ?= {
			add_progress = { value = -8 name = smd_devolution_bar }
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = smd_devolution.2.b
		var:smd_devolution_state ?= {
			add_modifier = {
				name = smd_regional_hardline
				days = short_modifier_time
			}
		}
		je:smd_je_devolution ?= {
			add_progress = { value = 8 name = smd_devolution_bar }
		}
		ai_chance = { base = 40 }
	}
}
