smd_devolution.4 = {
	type = country_event
	placement = ROOT
	title = smd_devolution.4.t
	desc = smd_devolution.4.d
	flavor = smd_devolution.4.f
	event_image = { video = "africa_public_protest" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
	}

	immediate = {
		var:smd_devolution_state ?= { save_scope_as = smd_devolution_state }
		if = {
			limit = { exists = var:smd_devolution_culture }
			var:smd_devolution_culture = { save_scope_as = smd_devolution_culture }
		}
		else_if = {
			limit = {
				exists = scope:smd_devolution_state
				scope:smd_devolution_state = {
					any_scope_culture = {
						NOT = { is_primary_culture_of = ROOT }
						has_homeland = PREV
					}
				}
			}
			scope:smd_devolution_state = {
				random_scope_culture = {
					limit = {
						NOT = { is_primary_culture_of = ROOT }
						has_homeland = PREV
					}
					save_scope_as = smd_devolution_culture
				}
			}
		}
	}

	option = {
		name = smd_devolution.4.a
		default_option = yes
		add_modifier = {
			name = smd_separatist_crisis
			days = normal_modifier_time
		}
		scope:smd_devolution_state ?= {
			add_radicals_in_state = {
				value = medium_radicals
			}
		}
		if = {
			limit = { exists = scope:smd_devolution_culture }
			create_political_movement = {
				type = movement_cultural_minority
				culture = scope:smd_devolution_culture
			}
		}
		ai_chance = { base = 100 }
	}
}
