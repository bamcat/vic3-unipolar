wild_nineties.20 = {
	type = country_event
	placement = root

	event_image = { video = "unspecific_sick_in_hospital" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	title = wild_nineties.20.t
	desc = wild_nineties.20.d
	left_icon = scope:rus_wn_yeltsin_scope

	duration = 3

	trigger = {
		rus_wn_yeltsin_in_power = yes
		NOT = { has_variable = rus_wn_succession_done }
	}

	immediate = {
		random_scope_character = {
			limit = { has_variable = rus_wn_yeltsin }
			save_scope_as = rus_wn_yeltsin_scope
		}
	}

	option = {
		name = wild_nineties.20.a
		default_option = yes
		random_scope_character = {
			limit = { has_variable = rus_wn_yeltsin }
			kill_character = yes
		}
		if = {
			limit = { any_scope_character = { has_role = character_role_prime_minister } }
			set_variable = { name = rus_wn_successor value = flag:pm }
		}
		else_if = {
			limit = { rus_wn_has_character = { T = RUS_viktor_chernomyrdin } }
			set_variable = { name = rus_wn_successor value = flag:chernomyrdin }
		}
		rus_wn_install_successor = yes
		rus_wn_record_successor = yes
	}
}
