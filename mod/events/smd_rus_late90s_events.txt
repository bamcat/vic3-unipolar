rus_l90.0 = {
	type = country_event
	hidden = yes

	trigger = {
		c:RUS ?= this
	}

	immediate = {
		if = {
			limit = {
				has_variable = chechen_war_started
				NOT = { has_variable = rus_l90_khasavyurt_fired }
				NOT = { has_variable = rus_l90_ichkeria_started }
				NOT = { has_journal_entry = je_first_chechen_war }
				rus_l90_ichkeria_independent = yes
				rus_l90_at_peace_with_ichkeria = yes
			}
			set_variable = rus_l90_khasavyurt_fired
			trigger_event = { id = rus_l90.1 popup = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = chechen_war_started }
				NOT = { has_global_variable = chechen_war_started }
				NOT = { has_variable = rus_l90_ichkeria_started }
				NOT = { has_global_variable = ichkeria_recognised }
				game_date >= 1997.1.1
				rus_l90_ichkeria_independent = yes
				rus_l90_at_peace_with_ichkeria = yes
			}
			rus_l90_unlock_ichkeria = { R = 15 }
			trigger_event = { id = rus_l90.7 popup = yes }
		}
		if = {
			limit = {
				has_variable = rus_l90_ichkeria_started
				NOT = { has_global_variable = rus_l90_cch_election_done }
				game_date >= 1997.1.1
				rus_l90_ichkeria_independent = yes
				rus_l90_at_peace_with_ichkeria = yes
			}
			set_global_variable = rus_l90_cch_election_done
			c:CCH = { trigger_event = { id = rus_l90.3 popup = yes } }
		}
		if = {
			limit = {
				has_variable = rus_l90_ichkeria_started
				NOT = { has_variable = rus_l90_dagestan_done }
				NOT = { has_variable = rus_l90_second_war }
				NOT = { has_global_variable = ichkeria_recognised }
				rus_l90_rad_above = { V = 65 }
				rus_l90_ichkeria_independent = yes
				rus_l90_at_peace_with_ichkeria = yes
				rus_l90_owns_dagestan = yes
			}
			random = {
				chance = 20
				set_variable = rus_l90_dagestan_done
				set_variable = { name = rus_l90_dagestan_recent days = 20 }
				trigger_event = { id = rus_l90.10 popup = yes }
			}
		}
		if = {
			limit = {
				has_variable = rus_l90_dagestan_done
				NOT = { has_variable = rus_l90_dagestan_recent }
				NOT = { has_variable = rus_l90_explosions_done }
				rus_l90_rad_above = { V = 50 }
			}
			random = {
				chance = 50
				set_variable = rus_l90_explosions_done
				trigger_event = { id = rus_l90.11 popup = yes }
			}
		}
		if = {
			limit = {
				has_variable = rus_wn_default_happened
				NOT = { has_variable = rus_l90_primakov_fired }
				NOT = { has_variable = rus_wn_succession_done }
				rus_wn_yeltsin_in_power = yes
				rus_wn_has_character = { T = RUS_yevgeny_primakov }
			}
			set_variable = rus_l90_primakov_fired
			trigger_event = { id = rus_l90.20 popup = yes }
		}
		if = {
			limit = {
				has_variable = rus_l90_primakov_govt
				NOT = { has_variable = rus_l90_primakov_recent }
				NOT = { has_variable = rus_l90_family_clash_fired }
				rus_l90_primakov_is_pm = yes
				rus_wn_yeltsin_in_power = yes
				rus_wn_family_above = { V = 40 }
			}
			set_variable = rus_l90_family_clash_fired
			trigger_event = { id = rus_l90.21 popup = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = rus_l90_governors_fired }
				NOT = { has_variable = rus_wn_succession_done }
				rus_wn_yeltsin_in_power = yes
				game_date >= 1999.6.1
				rus_wn_has_character = { T = RUS_yuri_luzhkov }
				OR = {
					has_variable = rus_l90_primakov_dismissed
					any_scope_character = {
						has_template = RUS_yuri_luzhkov
						has_variable = rus_wn_standing
						var:rus_wn_standing >= 25
					}
				}
			}
			set_variable = rus_l90_governors_fired
			trigger_event = { id = rus_l90.22 popup = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = rus_l90_impeachment_fired }
				NOT = { has_variable = rus_wn_succession_done }
				rus_wn_yeltsin_in_power = yes
				game_date >= 1998.9.1
				legitimacy < 50
				OR = {
					has_variable = rus_wn_default_happened
					has_variable = chechen_war_started
				}
			}
			random = {
				chance = 25
				set_variable = rus_l90_impeachment_fired
				trigger_event = { id = rus_l90.23 popup = yes }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = rus_l90_kosovo_fired }
				c:YUG ?= { has_variable = nato_bombing_started }
			}
			set_variable = rus_l90_kosovo_fired
			trigger_event = { id = rus_l90.30 popup = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = rus_l90_pristina_fired }
				has_global_variable = kosovo_under_international_administration
				is_at_war = no
			}
			set_variable = rus_l90_pristina_fired
			trigger_event = { id = rus_l90.31 popup = yes }
		}
	}
}
