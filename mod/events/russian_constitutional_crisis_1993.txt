rus_events.1 = {
	type = country_event
	placement = root

	event_image = { video = "parliament" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = rus_events.1.t
	desc = rus_events.1.d
	flavor = rus_events.1.f
	left_icon = scope:rus_khasbulatov_scope

	duration = 3

	immediate = {
		random_scope_character = {
			limit = { has_template = RUS_ruslan_khasbulatov }
			save_scope_as = rus_khasbulatov_scope
		}
	}

	option = {
		name = rus_events.1.a
		default_option = yes
		rus_wn_dvoe_change = { V = 3 }
	}

	option = {
		name = rus_events.1.b
		rus_wn_dvoe_change = { V = -2 }
		rus_wn_inst_change = { I = court V = 10 }
	}
}

rus_events.7 = {
	type = country_event
	placement = root

	event_image = { video = "stormingofwhitehouse" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	title = rus_events.7.t
	desc = rus_events.7.d
	left_icon = scope:rus_grachev_scope

	duration = 3

	trigger = {
		rus_wn_yeltsin_in_power = yes
		NOT = { has_variable = rus_1993_result }
	}

	immediate = {
		random_scope_character = {
			limit = { has_template = RUS_pavel_grachev }
			save_scope_as = rus_grachev_scope
		}
	}

	option = {
		name = rus_events.7.a
		default_option = yes
		trigger = {
			var:rus_inst_army >= 20
		}
		set_variable = { name = rus_1993_result value = flag:president }
		add_modifier = { name = rus_october_1993_storming months = 24 }
		rus_wn_add_standing = { T = RUS_pavel_grachev V = 10 }
		trigger_event = { id = rus_events.9 days = 60 popup = yes }
	}

	option = {
		name = rus_events.7.b
		trigger = {
			var:rus_inst_army <= -20
		}
		rus_wn_parliament_victory = yes
	}

	option = {
		name = rus_events.7.c
		trigger = {
			var:rus_inst_army > -20
			var:rus_inst_army < 20
		}
		set_variable = { name = rus_1993_result value = flag:stalemate }
		add_modifier = { name = rus_long_october months = 24 }
		trigger_event = { id = rus_events.9 days = 120 popup = yes }
	}
}
