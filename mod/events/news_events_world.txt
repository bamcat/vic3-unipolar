#  9. ELECTION IN LEADING POWER 
news_events_world.9 = {
	type = country_event
	placement = scope:news_event_state

	title = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:news_election_presidential_party }
				desc = news_events_world.9.t_presidential
			}
			triggered_desc = {
				trigger = { exists = scope:news_election_leading_party }
				desc = news_events_world.9.t_party
			}
			triggered_desc = {
				trigger = { exists = scope:news_election_government_party }
				desc = news_events_world.9.t_government
			}
			desc = news_events_world.9.t
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:news_election_presidential_party }
				desc = news_events_world.9.d_presidential
			}
			triggered_desc = {
				trigger = { exists = scope:news_election_leading_party }
				desc = news_events_world.9.d_party
			}
			triggered_desc = {
				trigger = { exists = scope:news_election_government_party }
				desc = news_events_world.9.d_government
			}
			desc = news_events_world.9.d
		}
	}
	flavor = news_events_world.9.f

	duration = 3
	gui_window = news_event_window

	event_image = { video = "southamerica_election" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/newspaper_icon.dds"

	trigger = { is_player = yes news_no_eur_subject = yes NOT = { has_variable = news_open_world_9 } NOT = { has_variable = news_off_world_9 } }

	immediate = {
		set_variable = { name = news_open_world_9 days = 90 }
		news_pick_title_variant = yes
		news_set_masthead_id = yes
		if = { limit = { exists = scope:news_new_ruler } set_variable = { name = news_world_9_new_ruler_ref value = scope:news_new_ruler months = 4 } }
		var:news_world_9_new_ruler_ref ?= { save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.

		# Resolve election-party information inside the newspaper event itself.
		# Party scopes proved unreliable when carried through the world-news
		# broadcast hop, while the source-country scope is stable.
		scope:news_source_country = {
			if = {
				limit = { has_law_or_variant = law_type:law_presidential_republic }
				ruler ?= {
					interest_group ?= {
						party ?= { save_scope_as = news_election_presidential_party }
					}
				}
			}

			# Best approximation of the electoral winner available to script:
			# the active party with the strongest campaign momentum at the
			# instant the election-end event is constructed.
			ordered_active_party = {
				limit = { party_has_members = yes }
				order_by = momentum
				position = 0
				check_range_bounds = no
				save_scope_as = news_election_leading_party
			}

			# Parliamentary / constitutional-monarchy fallback. If no campaign
			# party scope can be resolved, identify the party represented by the
			# strongest Interest Group currently in government. This guarantees
			# Britain-style HM's Governments get a party-specific newspaper
			# without pretending that the monarch personally won the election.
			if = {
				limit = { NOT = { exists = scope:news_election_leading_party } }
				ordered_interest_group = {
					limit = {
						is_in_government = yes
						party ?= { party_has_members = yes }
					}
					order_by = ig_clout
					position = 0
					check_range_bounds = no
					party ?= { save_scope_as = news_election_government_party }
				}
			}
		}
	
		if = { limit = { exists = scope:news_election_government_party } set_variable = { name = news_world_9_news_election_government_party_ref value = scope:news_election_government_party months = 4 } }
		var:news_world_9_news_election_government_party_ref ?= { save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
		if = { limit = { exists = scope:news_election_leading_party } set_variable = { name = news_world_9_news_election_leading_party_ref value = scope:news_election_leading_party months = 4 } }
		var:news_world_9_news_election_leading_party_ref ?= { save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
		if = { limit = { exists = scope:news_election_presidential_party } set_variable = { name = news_world_9_news_election_presidential_party_ref value = scope:news_election_presidential_party months = 4 } }
		var:news_world_9_news_election_presidential_party_ref ?= { save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
		if = { limit = { exists = scope:news_source_country } set_variable = { name = news_world_9_news_source_country_ref value = scope:news_source_country months = 4 } }
		var:news_world_9_news_source_country_ref ?= { save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
	}

	option = { name = news_events_world.9.a default_option = yes remove_variable = news_open_world_9 }
}
