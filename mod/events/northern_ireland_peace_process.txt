		c:GBR ?= this

		s:STATE_ULSTER.region_state:GBR = {
			add_loyalists_in_state = {
				value = 0.10
			}
		}
	}
}

ni_peace_events.5 = {
	type = country_event
	placement = ROOT
	title = ni_peace_events.5.t
	desc = ni_peace_events.5.d
	event_image = { video = "unspecific_signed_contract" }
	duration = 3

	trigger = {
		c:GBR ?= this
		exists = c:IRE
		NOT = { has_variable = smd_ni_withdrawal_cooldown }
		je:je_troubles_in_northern_ireland ?= {
			"scripted_bar_progress(troubles_troop_strength_bar)" <= 5
		}
	}

	immediate = {
		set_variable = {
			name = smd_ni_withdrawal_cooldown
			days = 180
		}
	}

	option = {
		name = ni_peace_events.5.a
		custom_tooltip = ni_peace_events.5.a.tt
		s:STATE_ULSTER.region_state:GBR = {
			set_state_owner = c:IRE
		}
		c:IRE = {
			trigger_event = { id = ni_peace_events.6 days = 2 }
		}
		ai_chance = {
			base = 5
		}
	}

	option = {
		name = smd_ni_refuse_withdrawal
		default_option = yes
		custom_tooltip = smd_ni_troop_presence_plus_20
		je:je_troubles_in_northern_ireland = {
			add_progress = {
				value = 20
				name = troubles_troop_strength_bar
			}
		}
		add_treasury = {
			value = gdp
			multiply = -0.005
		}
		s:STATE_ULSTER.region_state:GBR = {
			add_radicals_in_state = {
				value = 0.10
				culture = cu:irish
			}
		}
		ai_chance = {
			base = 95
		}
	}
}
