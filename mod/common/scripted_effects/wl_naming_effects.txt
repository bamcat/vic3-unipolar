wl_reclassify = {
	save_scope_as = wl_slot
	if = {
		limit = {
			has_variable = wl_type
			var:wl_type = 7
			exists = var:wl_play
			NOT = { has_variable = wl_subj_civil }
		}
		var:wl_play = { save_scope_as = wl_play }
		scope:wl_war = {
			attacker_warleader = { save_scope_as = wl_att }
			defender_warleader = { save_scope_as = wl_def }
		}
		set_variable = { name = wl_x_old value = var:wl_x }
		wl_set_independence_side = yes
		if = {
			limit = {
				NOT = { var:wl_x = var:wl_x_old }
			}
			wl_set_keys = yes
			wl_assign_ordinal = yes
		}
		remove_variable = wl_x_old
	}
	if = {
		limit = {
			NOT = { has_variable = wl_locked }
			wl_is_great_war_size = yes
		}
		if = {
			limit = { var:wl_year >= 1900 }
			wl_set_type = { T = 2 }
		}
		else = {
			wl_set_type = { T = 3 }
		}
		wl_set_keys = yes
		wl_assign_ordinal = yes
	}
	else_if = {
		limit = {
			NOT = { has_variable = wl_locked }
			var:wl_type = 1
			wl_total_participants >= 5
		}
		wl_set_type = { T = 14 }
		wl_set_keys = yes
		wl_assign_ordinal = yes
	}
}

wl_name_major_war = {
	clear_saved_scope = wl_bs
	clear_saved_scope = wl_g1
	clear_saved_scope = wl_g2
	scope:wl_war = {
		every_war_participant = {
			limit = { country_rank = rank_value:great_power }
			save_scope_as = wl_g1
			if = {
				limit = {
					NOT = { exists = scope:wl_bs }
					NOT = { scope:wl_war = { is_on_defending_side = scope:wl_g1 } }
				}
				scope:wl_war = {
					every_war_participant = {
						limit = {
							country_rank = rank_value:great_power
							is_adjacent_to_country = scope:wl_g1
						}
						save_scope_as = wl_g2
						if = {
							limit = {
								NOT = { exists = scope:wl_bs }
								scope:wl_war = { is_on_defending_side = scope:wl_g2 }
							}
							random_scope_state = {
								limit = {
									any_neighbouring_state = { owner ?= scope:wl_g1 }
								}
								save_scope_as = wl_bs
							}
						}
					}
				}
			}
		}
	}
	if = {
		limit = { exists = scope:wl_bs }
		set_variable = { name = wl_region value = scope:wl_bs.region }
		wl_set_type = { T = 14 }
	}
	else = {
		scope:wl_def = {
			random_scope_state = {
				limit = {
					any_neighbouring_state = {
						owner ?= { has_war_with = scope:wl_def }
					}
				}
				save_scope_as = wl_bs
			}
		}
		if = {
			limit = { exists = scope:wl_bs }
			set_variable = { name = wl_st value = scope:wl_bs.state_region }
			wl_set_type = { T = 27 }
		}
		else = {
			if = {
				limit = { var:wl_gp_a >= var:wl_gp_b }
				set_variable = { name = wl_flavor value = var:wl_n_a }
			}
			else = {
				set_variable = { name = wl_flavor value = var:wl_n_b }
			}
			if = {
				limit = { var:wl_flavor > 7 }
				set_variable = { name = wl_flavor value = 7 }
			}
			if = {
				limit = { var:wl_flavor < 2 }
				set_variable = { name = wl_flavor value = 2 }
			}
			wl_set_type = { T = 28 }
		}
	}
	clear_saved_scope = wl_bs
	clear_saved_scope = wl_g1
	clear_saved_scope = wl_g2
}

wl_set_independence_side = {
	if = {
		limit = { scope:wl_att = { is_subject_of = scope:wl_def } }
		set_variable = { name = wl_x value = scope:wl_att.country_definition }
	}
	else_if = {
		limit = { scope:wl_def = { is_subject_of = scope:wl_att } }
		set_variable = { name = wl_x value = scope:wl_def.country_definition }
	}
	else_if = {
		limit = {
			exists = scope:wl_play.initiator
			exists = scope:wl_play.target
			NOT = { scope:wl_play.initiator = scope:wl_play.target }
		}
		set_variable = { name = wl_x value = scope:wl_play.initiator.country_definition }
	}
	else_if = {
		limit = { has_variable = wl_rev }
		set_variable = { name = wl_x value = var:wl_rev }
	}
}
