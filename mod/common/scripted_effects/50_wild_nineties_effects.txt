rus_wn_clear_pm = {
	every_scope_character = {
		limit = { has_role = character_role_prime_minister }
		remove_character_role = character_role_prime_minister
	}
	if = { limit = { has_modifier = rus_pm_gaidar } remove_modifier = rus_pm_gaidar }
	if = { limit = { has_modifier = rus_pm_chernomyrdin } remove_modifier = rus_pm_chernomyrdin }
	if = { limit = { has_modifier = rus_pm_kiriyenko } remove_modifier = rus_pm_kiriyenko }
	if = { limit = { has_modifier = rus_pm_primakov } remove_modifier = rus_pm_primakov }
	if = { limit = { has_modifier = rus_pm_stepashin } remove_modifier = rus_pm_stepashin }
	if = { limit = { has_modifier = rus_pm_putin } remove_modifier = rus_pm_putin }
	if = { limit = { has_modifier = rus_pm_nemtsov } remove_modifier = rus_pm_nemtsov }
	if = { limit = { has_modifier = rus_pm_luzhkov } remove_modifier = rus_pm_luzhkov }
	if = { limit = { has_modifier = rus_pm_lebed } remove_modifier = rus_pm_lebed }
}

rus_wn_appoint_pm = {
	rus_wn_clear_pm = yes
	random_scope_character = {
		limit = {
			has_template = $T$
			is_character_alive = yes
		}
		add_character_role = character_role_prime_minister
		if = {
			limit = { NOT = { has_variable = rus_wn_standing } }
			set_variable = { name = rus_wn_standing value = 0 }
		}
		change_variable = { name = rus_wn_standing add = 20 }
		save_scope_as = rus_wn_new_pm
		owner = { pm_announce = { PM = scope:rus_wn_new_pm } }
	}
	add_modifier = { name = $M$ }
	set_variable = { name = rus_wn_pm_blame value = 0 }
}

rus_wn_parliament_victory = {
	set_variable = { name = rus_1993_result value = flag:parliament }
	if = {
		limit = { rus_wn_has_character = { T = RUS_alexander_rutskoy } }
		rus_wn_make_ruler = { T = RUS_alexander_rutskoy }
	}
	else_if = {
		limit = { rus_wn_has_character = { T = RUS_ruslan_khasbulatov } }
		rus_wn_make_ruler = { T = RUS_ruslan_khasbulatov }
	}
	else = {
		rus_wn_opposition_takes_power = yes
	}
	activate_law = law_type:law_parliamentary_republic
	active_law:lawgroup_governance_principles ?= {
		add_amendment = {
			type = amendment_rus_parliamentary_oversight
			sponsor = root.ig:ig_rural_folk
		}
	}
	add_modifier = { name = rus_parliament_victory months = 36 }
	rus_wn_add_standing = { T = RUS_ruslan_khasbulatov V = 30 }
}

rus_wn_compromise_constitution = {
	set_variable = { name = rus_1993_result value = flag:compromise }
	set_variable = rus_1993_constitution
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_presidential_republic
				has_law_or_variant = law_type:law_parliamentary_republic
			}
		}
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_rus_balanced_constitution
				sponsor = root.ig:ig_intelligentsia
			}
		}
	}
	add_modifier = { name = rus_zero_option months = 24 }
}

rus_wn_install_successor = {
	custom_tooltip = rus_wn_install_successor_tt
	hidden_effect = {
		if = {
			limit = { var:rus_wn_successor ?= flag:putin }
			rus_wn_make_ruler = { T = RUS_vladimir_putin }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:stepashin }
			rus_wn_make_ruler = { T = RUS_sergei_stepashin }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:primakov }
			rus_wn_make_ruler = { T = RUS_yevgeny_primakov }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:luzhkov }
			rus_wn_make_ruler = { T = RUS_yuri_luzhkov }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:chernomyrdin }
			rus_wn_make_ruler = { T = RUS_viktor_chernomyrdin }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:nemtsov }
			rus_wn_make_ruler = { T = RUS_boris_nemtsov }
		}
		else_if = {
			limit = { var:rus_wn_successor ?= flag:lebed }
			rus_wn_make_ruler = { T = RUS_alexander_lebed }
		}
		if = {
			limit = {
				OR = {
					rus_wn_yeltsin_in_power = yes
					var:rus_wn_successor ?= flag:pm
				}
				any_scope_character = {
					has_role = character_role_prime_minister
					is_character_alive = yes
				}
			}
			random_scope_character = {
				limit = { has_role = character_role_prime_minister }
				remove_character_role = character_role_prime_minister
				set_character_as_ruler = yes
			}
			set_variable = { name = rus_wn_successor value = flag:pm }
		}
		if = {
			limit = { rus_wn_yeltsin_in_power = no }
			rus_wn_successor_modifier = yes
			set_variable = rus_wn_succession_done
		}
	}
}

rus_wn_opposition_takes_power = {
	if = {
		limit = {
			ig:ig_rural_folk ?= {
				leader ?= {
					is_character_alive = yes
					NOT = { has_variable = rus_wn_yeltsin }
				}
			}
		}
		ig:ig_rural_folk.leader = {
			if = {
				limit = { has_role = character_role_prime_minister }
				remove_character_role = character_role_prime_minister
			}
			set_character_as_ruler = yes
		}
	}
	else_if = {
		limit = {
			ig:ig_trade_unions ?= {
				leader ?= {
					is_character_alive = yes
					NOT = { has_variable = rus_wn_yeltsin }
				}
			}
		}
		ig:ig_trade_unions.leader = {
			if = {
				limit = { has_role = character_role_prime_minister }
				remove_character_role = character_role_prime_minister
			}
			set_character_as_ruler = yes
		}
	}
}

rus_wn_found_sibneft = {
	if = {
		limit = {
			NOT = { has_company = company_type:company_sibneft }
			can_establish_company = yes
		}
		add_company = company_type:company_sibneft
		company:company_sibneft ?= {
			set_company_establishment_date = 1995.8.24
		}
		if = {
			limit = {
				any_scope_state = { state_region = s:STATE_OB }
			}
			company:company_sibneft ?= {
				set_company_state_region = s:STATE_OB
			}
		}
		else = {
			company:company_sibneft ?= {
				set_company_state_region = root.capital.state_region
			}
		}
	}
}
