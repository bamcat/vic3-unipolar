fin_apply_monetary_stance = {
	if = { limit = { has_modifier = fin_monetary_stance_1 } remove_modifier = fin_monetary_stance_1 }
	if = { limit = { has_modifier = fin_monetary_stance_2 } remove_modifier = fin_monetary_stance_2 }
	if = { limit = { has_modifier = fin_monetary_stance_4 } remove_modifier = fin_monetary_stance_4 }
	if = { limit = { has_modifier = fin_monetary_stance_5 } remove_modifier = fin_monetary_stance_5 }
	if = {
		limit = { var:cb_fiscal_policy ?= 1 }
		add_modifier = { name = fin_monetary_stance_1 }
	}
	else_if = {
		limit = { var:cb_fiscal_policy ?= 2 }

	else_if = {

	else_if = {
		limit = { var:cb_fiscal_policy ?= 5 }

			}
		}
		else_if = {
			limit = { has_law = law_type:law_full_reserve_banking }
			if = {
				limit = { var:cb_fiscal_policy = 1 }
				add_modifier = { name = cb_policy_full_1 months = -1 }
			}
			else_if = {
				limit = { var:cb_fiscal_policy = 2 }
				add_modifier = { name = cb_policy_full_2 months = -1 }
			}
			else_if = {
				limit = { var:cb_fiscal_policy = 4 }
				add_modifier = { name = cb_policy_full_4 months = -1 }
			}
			else_if = {
				limit = { var:cb_fiscal_policy = 5 }
				add_modifier = { name = cb_policy_full_5 months = -1 }
			}
		}
		fin_apply_monetary_stance = yes
	}
}

	custom_description = {
		text = fin_capital_flight_change
		value = $V$

			limit = { NOT = { has_variable = fin_capital_flight } }
			set_variable = { name = fin_capital_flight value = 0 }

fin_apply_capital_flight = {
	if = { limit = { has_modifier = fin_capital_flight_outflow } remove_modifier = fin_capital_flight_outflow }
	if = {
		limit = { var:fin_capital_flight > 0 }
		add_modifier = {
			name = fin_capital_flight_outflow
			multiplier = var:fin_capital_flight
		}
	}
	if = { limit = { has_modifier = fin_capital_flight_1 } remove_modifier = fin_capital_flight_1 }
	if = { limit = { has_modifier = fin_capital_flight_2 } remove_modifier = fin_capital_flight_2 }
	if = { limit = { has_modifier = fin_capital_flight_3 } remove_modifier = fin_capital_flight_3 }
	if = {
		limit = { var:fin_capital_flight >= 70 }
		add_modifier = { name = fin_capital_flight_3 }
	}
	else_if = {
		limit = { var:fin_capital_flight >= 45 }
		add_modifier = { name = fin_capital_flight_2 }
	}
	else_if = {
		limit = { var:fin_capital_flight >= 20 }
		add_modifier = { name = fin_capital_flight_1 }
	}
}

		limit = { has_variable = fin_bond_stock }

			limit = { has_variable = fin_bonds_issuing }
			add_treasury = fin_bond_issue_amount
			change_variable = { name = fin_bond_stock add = fin_bond_issue_amount }
		}
		else = {
			change_variable = { name = fin_bond_stock multiply = 0.97 }
		}
	}
	set_variable = { name = fin_bond_ratio_var value = fin_bond_ratio }
	set_variable = { name = fin_debt_ratio_var value = fin_debt_ratio }
	fin_apply_bond_interest = yes
}

fin_monthly_update = {
	fin_update_cpi = yes
	fin_update_bonds = yes
	if = {
		limit = { NOT = { has_variable = fin_capital_flight } }
		set_variable = { name = fin_capital_flight value = 0 }
	}
	if = {
		limit = { var:fin_cpi >= 30 }
		fin_change_capital_flight = { V = 0.5 }
	}
	if = {
		limit = { var:fin_bond_ratio_var >= 10 }
		fin_change_capital_flight = { V = 0.5 }
	}
	if = {
		limit = {
			var:fin_bond_ratio_var < 10
			var:fin_cpi < 15
		}
		fin_change_capital_flight = { V = -1 }
	}
	else_if = {
		limit = {
			var:fin_bond_ratio_var < 10
			var:fin_cpi < 30
		}
		fin_change_capital_flight = { V = -0.25 }
	}
	fin_apply_capital_flight = yes
}
