smd_effect_advance_enrichment = {
	if = {
		limit = {
			has_variable = smd_enrichment_active
			NOT = { has_variable = smd_enriched }
		}
		change_variable = {
			name = smd_enrichment_progress
			add = 4
		}
		every_scope_state = {
			limit = {
				has_building = building_nuclear_plant
			}
			root = {
				change_variable = {
					name = smd_enrichment_progress
					add = 0.1
				}
			}
		}
		change_variable = {
			name = smd_enrichment_progress
			add = -3
		}
		if = {
			limit = { var:smd_enrichment_progress < 4 }
			set_variable = {
				name = smd_enrichment_progress
				value = 4
			}
		}
		if = {
			limit = { var:smd_enrichment_progress >= 100 }
			set_variable = {
				name = smd_enrichment_progress
				value = 100
			}
		}
	}
}

# Advance production by one month (2x enrichment rate)
smd_effect_advance_production = {
	if = {
		limit = {
			has_variable = smd_production_active
			has_variable = smd_enriched
		}
		change_variable = {
			name = smd_production_progress
			add = 8
		}
		every_scope_state = {
			limit = {
				has_building = building_nuclear_plant
			}
			owner = {
				change_variable = {
					name = smd_production_progress
					add = 0.1
				}
			}
		}
		change_variable = {
			name = smd_production_progress
			add = -3
		}
		if = {
			limit = { var:smd_production_progress >= 100 }
			set_variable = {
				name = smd_production_progress
				value = 100
			}
		}
	}
}

# Begin enrichment
smd_effect_begin_enrichment = {
	set_variable = smd_enrichment_active
	if = {
		limit = { NOT = { has_variable = smd_enrichment_progress } }
		set_variable = {
			name = smd_enrichment_progress
			value = 0
		}
	}
}

# Complete enrichment
smd_effect_complete_enrichment = {
	remove_variable = smd_enrichment_active
	set_variable = smd_enriched
	trigger_event = {
		id = smd_nuclear_events.2
		popup = yes
	}
}

# Begin weapon production
smd_effect_begin_production = {
	set_variable = smd_production_active
	if = {
		limit = { NOT = { has_variable = smd_production_progress } }
		set_variable = {
			name = smd_production_progress
			value = 0
		}
	}
}

# Complete production — +1 warhead, reset progress
smd_effect_complete_production = {
	remove_variable = smd_production_active
	set_variable = {
		name = smd_production_progress
		value = 0
	}
	if = {
		limit = { NOT = { has_variable = smd_nukes } }
		set_variable = {
			name = smd_nukes
			value = 0
		}
	}
	change_variable = {
		name = smd_nukes
		add = 1
	}
	trigger_event = {
		id = smd_nuclear_events.3
		popup = yes
	}
}

nuclear_rebuild_enemy_list = {
	save_scope_as = smd_wmd_owner
	clear_variable_list = nuclear_enemy_list
	clear_variable_list = nuclear_enemy_states_list
	clear_variable_list = nuclear_ranged_states_list
 
	if = {
		limit = { is_at_war = yes }
		every_country = {
			limit = {
				has_war_with = scope:smd_wmd_owner
				is_subject = no
				country_rank >= rank_value:unrecognized
			}
			scope:smd_wmd_owner = {
				add_to_variable_list = {
					name = nuclear_enemy_list
					target = PREV
				}
			}
			every_scope_state = {
				limit = { owner = PREV 
					OR = {
						any_neighbouring_state = {
							owner = scope:smd_wmd_owner
						}
						any_neighbouring_state = {
							owner = {
								any_scope_treaty = {
									any_scope_article = {
										has_type = host_military_bases
										source_country = scope:smd_wmd_owner
									}
								}
							}
						}
						is_blockaded_by = scope:smd_wmd_owner
					}
				}
				scope:smd_wmd_owner = {
					add_to_variable_list = {
						name = nuclear_enemy_states_list
						target = PREV
					}
				}
			}
			every_scope_state = {
				limit = {
					owner = PREV
					OR = {
						any_neighbouring_state = {
							owner = scope:smd_wmd_owner
						}
						any_neighbouring_state = {
							owner = {
								any_scope_treaty = {
									any_scope_article = {
										has_type = host_military_bases
										source_country = scope:smd_wmd_owner
									}
								}
							}
						}
						is_blockaded_by = scope:smd_wmd_owner
						scope:smd_wmd_owner = { has_variable = icbm_programme_complete }
					}
				}
				scope:smd_wmd_owner = {
					add_to_variable_list = {
						name = nuclear_ranged_states_list
						target = PREV
					}
				}
			}
		}
	}
 
	every_in_list = {
		variable = wmd_target_list
		limit = {
			OR = {
				owner = { NOT = { has_war_with = scope:smd_wmd_owner } }
				AND = {
					NOT = { scope:smd_wmd_owner = { has_variable = icbm_programme_complete } }
					NOT = {
						any_neighbouring_state = {
							owner = scope:smd_wmd_owner
						}
					}
				}
			}
		}
		scope:smd_wmd_owner = {
			remove_list_variable = {
				name = wmd_target_list
				target = PREV
			}
		}
	}
	every_in_list = {
		variable = wmd_target_list_c
		limit = {
			NOT = {
				any_in_list = {
					variable = wmd_target_list
					owner = THIS
				}
			}
		}
		scope:smd_wmd_owner = {
			remove_list_variable = {
				name = wmd_target_list_c
				target = PREV
			}
		}
	}
 
	every_in_list = {
		variable = conventional_target_list
		limit = {
			owner = { NOT = { has_war_with = scope:smd_wmd_owner } }
		}
		scope:smd_wmd_owner = {
			remove_list_variable = {
				name = conventional_target_list
				target = PREV
			}
		}
	}
	every_in_list = {
		variable = conventional_target_list_c
		limit = {
			NOT = {
				any_in_list = {
					variable = conventional_target_list
					owner = THIS
				}
			}
		}
		scope:smd_wmd_owner = {
			remove_list_variable = {
				name = conventional_target_list_c
				target = PREV
			}
		}
	}
 
	every_in_list = {
		variable = bombing_target_list
		limit = {
			owner = { NOT = { has_war_with = scope:smd_wmd_owner } }
		}
		scope:smd_wmd_owner = {
			remove_list_variable = {
				name = bombing_target_list
				target = PREV
			}
		}
	}
 
	nuclear_update_kill_rate_cache = yes
}

conventional_missile_strike = {
	save_scope_value_as = { name = smd_missile_devastation value = smd_conv_devastation_per_target_sv }
	save_scope_value_as = { name = smd_missile_casualties value = smd_conv_casualty_percent_per_target_sv }
	save_scope_value_as = { name = smd_missiles_per_target value = smd_conv_missiles_per_target_sv }
	change_variable = {
		name = conventional_missiles
		subtract = conventional_missiles_to_use_sv
	}
	if = {
		limit = { var:conventional_missiles < 0 }
		set_variable = { name = conventional_missiles value = 0 }
	}

	every_in_list = {
		variable = conventional_target_list
		add_devastation = scope:smd_missile_devastation
		kill_population_percent_in_state = { percent = scope:smd_missile_casualties }
		if = {
			limit = { scope:smd_missiles_per_target >= 10 }
			smd_missile_building_strikes = { HIGH = 30 LOW = 20 }
		}
		else_if = {
			limit = { scope:smd_missiles_per_target >= 5 }
			smd_missile_building_strikes = { HIGH = 18 LOW = 12 }
		}
		else = {
			smd_missile_building_strikes = { HIGH = 10 LOW = 7 }
		}
	}
}
