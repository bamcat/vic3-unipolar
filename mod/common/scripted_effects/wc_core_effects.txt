wc_init = {
	if = {
		limit = { NOT = { has_global_variable = wc_initialized } }
		set_global_variable = wc_initialized
		set_global_variable = { name = wc_next value = 1 }
		set_global_variable = { name = wc_months value = 0 }
		every_state_region = {
			add_to_global_variable_list = { name = wc_free target = this }
		}
		wc_update_year = yes
	}
}

wc_update_year = {
	set_global_variable = { name = wc_year value = 1700 }
	wc_year_step = { S = 256 }
	wc_year_step = { S = 128 }
	wc_year_step = { S = 64 }
	wc_year_step = { S = 32 }
	wc_year_step = { S = 16 }
	wc_year_step = { S = 8 }
	wc_year_step = { S = 4 }
	wc_year_step = { S = 2 }
	wc_year_step = { S = 1 }
}

wc_year_step = {
	change_global_variable = { name = wc_year add = $S$ }
	if = {
		limit = { NOT = { year >= global_var:wc_year } }
		change_global_variable = { name = wc_year subtract = $S$ }
	}
}

wc_monthly = {
	wc_init = yes
	change_global_variable = { name = wc_months add = 1 }
	if = {
		limit = { NOT = { has_global_variable = wc_tick_alive } }
		wc_start_ticker = yes
	}
}

wc_start_ticker = {
	set_global_variable = { name = wc_tick_alive days = 10 }
	random_country = {
		limit = { NOT = { is_country_type = decentralized } }
		trigger_event = { id = wc_events.1 days = 7 }
	}
}

wc_weekly = {
	wc_init = yes
	wc_update_year = yes
	every_country = {
		limit = { any_scope_war = { always = yes } }
		every_scope_war = {
			wc_record_war = yes
		}
	}
	clear_global_variable_list = wc_finished
	every_in_global_list = {
		variable = wc_active
		save_scope_as = wc_slot
		if = {
			limit = { exists = var:wc_war }
			var:wc_war = { save_scope_as = wc_war }
			wc_scan_sides = yes
			wc_measure = yes
			wc_reclassify = yes
		}
		else = {
			add_to_global_variable_list = { name = wc_finished target = this }
		}
	}
	every_in_global_list = {
		variable = wc_finished
		wc_finish = yes
	}
	clear_global_variable_list = wc_finished
	wc_rebuild_display = yes
}

wc_take_slot = {
	clear_saved_scope = wc_slot
	if = {
		limit = {
			any_in_global_list = {
				variable = wc_free
				always = yes
			}
		}
		random_in_global_list = {
			variable = wc_free
			save_scope_as = wc_slot
		}
		remove_list_global_variable = { name = wc_free target = scope:wc_slot }
	}
	else = {
		ordered_in_global_list = {
			variable = wc_records
			limit = { has_variable = wc_ended }
			order_by = wc_age
			position = 0
			check_range_bounds = no
			save_scope_as = wc_slot
		}
		if = {
			limit = { exists = scope:wc_slot }
			remove_list_global_variable = { name = wc_records target = scope:wc_slot }
			scope:wc_slot = { wc_clear_slot = yes }
		}
	}
}

wc_record_war = {
	clear_saved_scope = wc_war
	clear_saved_scope = wc_play
	clear_saved_scope = wc_slot
	clear_saved_scope = wc_att
	clear_saved_scope = wc_def
	save_scope_as = wc_war
	diplomatic_play ?= { save_scope_as = wc_play }
	if = {
		limit = {
			exists = scope:wc_play
			scope:wc_play = { NOT = { has_variable = wc_slot } }
			exists = attacker_warleader
			exists = defender_warleader
		}
		wc_init = yes
		wc_take_slot = yes
		if = {
			limit = { exists = scope:wc_slot }
			attacker_warleader = { save_scope_as = wc_att }
			defender_warleader = { save_scope_as = wc_def }
			scope:wc_play = { set_variable = { name = wc_slot value = scope:wc_slot } }
			scope:wc_slot = {
				set_variable = wc_used
				set_variable = { name = wc_index value = global_var:wc_next }
				set_variable = { name = wc_war value = scope:wc_war }
				set_variable = { name = wc_play value = scope:wc_play }
				set_variable = { name = wc_a value = scope:wc_att.country_definition }
				set_variable = { name = wc_b value = scope:wc_def.country_definition }
				set_variable = { name = wc_year value = global_var:wc_year }
				set_variable = { name = wc_start_m value = global_var:wc_months }
				set_variable = { name = wc_cas value = 0 }
				set_variable = { name = wc_dead value = 0 }
				set_variable = { name = wc_cas_a value = 0 }
				set_variable = { name = wc_cas_b value = 0 }
				set_variable = { name = wc_dead_a value = 0 }
				set_variable = { name = wc_dead_b value = 0 }
				set_variable = { name = wc_region value = scope:wc_play.region }
			}
			change_global_variable = { name = wc_next add = 1 }
			add_to_global_variable_list = { name = wc_records target = scope:wc_slot }
			add_to_global_variable_list = { name = wc_active target = scope:wc_slot }
			scope:wc_slot = {
				wc_scan_sides = yes
				wc_classify = yes
				wc_assign_ordinal = yes
			}
			wc_rebuild_display = yes
		}
	}
}

wc_scan_sides = {
	save_scope_as = wc_slot
	set_variable = { name = wc_n_a value = 0 }
	set_variable = { name = wc_n_b value = 0 }
	set_variable = { name = wc_gp_a value = 0 }
	set_variable = { name = wc_gp_b value = 0 }
	set_variable = { name = wc_sup_n_a value = 0 }
	set_variable = { name = wc_sup_n_b value = 0 }
	clear_variable_list = wc_sup_a_c
	clear_variable_list = wc_sup_b_c
	scope:wc_war = {
		attacker_warleader = { save_scope_as = wc_lead_a }
		defender_warleader = { save_scope_as = wc_lead_b }
		every_war_participant = {
			save_scope_as = wc_p
			if = {
				limit = { scope:wc_war = { is_on_defending_side = scope:wc_p } }
				scope:wc_slot = {
					change_variable = { name = wc_n_b add = 1 }
					if = {
						limit = { scope:wc_p = { country_rank = rank_value:great_power } }
						change_variable = { name = wc_gp_b add = 1 }
					}
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = wc_side_b target = scope:wc_p.country_definition } } }
						add_to_variable_list = { name = wc_side_b target = scope:wc_p.country_definition }
					}
					if = {
						limit = { scope:wc_p = { this = scope:wc_lead_b } }
						set_variable = { name = wc_lead_b_c value = scope:wc_p }
					}
					else = {
						change_variable = { name = wc_sup_n_b add = 1 }
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = wc_sup_b target = scope:wc_p.country_definition } } }
							add_to_variable_list = { name = wc_sup_b target = scope:wc_p.country_definition }
						}
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = wc_sup_b_c target = scope:wc_p } } }
							add_to_variable_list = { name = wc_sup_b_c target = scope:wc_p }
						}
					}
				}
			}
			else = {
				scope:wc_slot = {
					change_variable = { name = wc_n_a add = 1 }
					if = {
						limit = { scope:wc_p = { country_rank = rank_value:great_power } }
						change_variable = { name = wc_gp_a add = 1 }
					}
					if = {
						limit = { NOT = { is_target_in_variable_list = { name = wc_side_a target = scope:wc_p.country_definition } } }
						add_to_variable_list = { name = wc_side_a target = scope:wc_p.country_definition }
					}
					if = {
						limit = { scope:wc_p = { this = scope:wc_lead_a } }
						set_variable = { name = wc_lead_a_c value = scope:wc_p }
					}
					else = {
						change_variable = { name = wc_sup_n_a add = 1 }
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = wc_sup_a target = scope:wc_p.country_definition } } }
							add_to_variable_list = { name = wc_sup_a target = scope:wc_p.country_definition }
						}
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = wc_sup_a_c target = scope:wc_p } } }
							add_to_variable_list = { name = wc_sup_a_c target = scope:wc_p }
						}
					}
				}
			}
		}
	}
}

wc_set_outcome_winner = {
	clear_saved_scope = wc_slot
	scope:diplomatic_play ?= {
		var:wc_slot ?= { save_scope_as = wc_slot }
	}
	if = {
		limit = { exists = scope:wc_slot }
		scope:war ?= {
			save_scope_as = wc_war
			scope:wc_slot = { wc_measure = yes }
		}
		scope:wc_slot = {
			if = {
				limit = { is_target_in_variable_list = { name = wc_side_a target = root.country_definition } }
				set_variable = { name = wc_outcome value = 1 }
			}
			else = {
				set_variable = { name = wc_outcome value = 2 }
			}
			if = {
				limit = { has_variable = wc_ended }
				wc_rename_at_end = yes
			}
		}
	}
}

wc_finish = {
	if = {
		limit = { NOT = { has_variable = wc_ended } }
		set_variable = wc_ended
		set_variable = { name = wc_end_year value = global_var:wc_year }
		set_variable = { name = wc_end_m value = global_var:wc_months }
		if = {
			limit = { NOT = { has_variable = wc_outcome } }
			set_variable = { name = wc_outcome value = 3 }
		}
		remove_list_global_variable = { name = wc_active target = this }
		remove_variable = wc_war
		remove_variable = wc_play
		wc_rename_at_end = yes
		wc_rebuild_display = yes
	}
}

	every_in_global_list = {

		wc_refresh_lists = yes

	wc_fill_past_sorted = { NAME = wc_past_dur ORDER = wc_order_duration }
	wc_fill_past_sorted = { NAME = wc_past_old ORDER = wc_age }

		save_scope_as = wc_me

			variable = wc_records

}

					has_variable = wc_gp_b

				}
			}
		}

		}
		add_to_variable_list = { name = wc_war_l target = var:wc_war }
