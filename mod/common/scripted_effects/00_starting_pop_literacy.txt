# Very high literacy

# Examples: Sweden, Netherlands
REPLACE_OR_CREATE:effect_starting_pop_literacy_very_high = {
	set_institution_investment_level = {
			institution = institution_schools
			level = 5		
		}
	every_scope_pop = {
		set_pop_literacy = {
			literacy_rate = {
				value = 0.6
				if = {
					limit = {
						OR = {
							is_pop_type = bureaucrats
							is_pop_type = academics
							is_pop_type = clergymen
							is_pop_type = engineers
						}
					}
					add = 0.2
				}
				if = {
					limit = {
						strata = upper
					}
					add = 0.15
				}
				if = {
					limit = {
						OR = {
							is_pop_type = clerks
							is_pop_type = professionals
							is_pop_type = shopkeepers
						}
					}
					add = 0.1
				}
				if = {
					limit = {
						OR = {
							pop_employment_building_group = bg_subsistence_agriculture
							pop_employment_building_group = bg_subsistence_ranching
						}
					}
					add = -0.1
				}
				if = {
					limit = {
						state = { is_incorporated = no }
					}
					multiply = 0.5
				}
			}
		}
	}
}
