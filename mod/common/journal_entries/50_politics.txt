smd_je_democratic_consolidation = {
	icon = "gfx/interface/icons/event_icons/event_election.dds"

	group = je_group_domestic_policy

	is_shown_when_inactive = {
		has_variable = smd_generic_ready
		NOT = { is_country_type = decentralized }
		has_variable = smd_new_democracy
		NOT = { has_variable = smd_democracy_consolidated }
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_universal_suffrage
			has_law_or_variant = law_type:law_census_voting
		}
	}

	immediate = {
		remove_variable = smd_dc_election_held
		trigger_event = { id = smd_democracy.1 popup = yes }
	}

	scripted_progress_bar = smd_democracy_bar

	scripted_button = smd_btn_dem_judicial_reform
	scripted_button = smd_btn_dem_civic_education

	on_monthly_pulse = {
		random_events = {
			100 = 0
			3 = smd_democracy.3
			3 = smd_democracy.4
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = smd_je_democratic_consolidation_success_desc
		trigger = {
			custom_tooltip = {
				text = smd_je_democratic_consolidation_election_tt
				has_variable = smd_dc_election_held
			}
			custom_tooltip = {
				text = smd_je_democratic_consolidation_bar_half_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(smd_democracy_bar)" >= 50
				}
			}
		}
		effect = {
			trigger_event = { id = smd_democracy.5 popup = yes }
			remove_variable = smd_new_democracy
			set_variable = smd_democracy_consolidated
			remove_variable = smd_dc_election_held
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = smd_je_democratic_consolidation_failure_desc
		trigger = {
			custom_tooltip = {
				text = smd_je_democratic_consolidation_failure_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(smd_democracy_bar)" <= 0
				}
			}
		}
		effect = {
			trigger_event = { id = smd_democracy.6 popup = yes }
			remove_variable = smd_new_democracy
			set_variable = { name = smd_backslid days = 3650 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = smd_je_democratic_consolidation_invalid_desc
		trigger = {
			NOR = {
				has_law_or_variant = law_type:law_universal_suffrage
				has_law_or_variant = law_type:law_census_voting
			}
		}
		effect = {
			remove_variable = smd_new_democracy
			set_variable = smd_was_undemocratic
		}
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
