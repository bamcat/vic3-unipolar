je_rwandan_genocide = {
    icon = "gfx/interface/icons/event_icons/event_skull.dds"
    group = je_group_dilemmas
    should_be_pinned_by_default_uninvolved_or_context = yes

    is_shown_when_inactive = {
        c:RWA ?= this
        game_date >= 1994.04.01
        game_date <= 1994.12.31
        NOT = { has_variable = genocide_ended }
    }

    possible = {
        c:RWA ?= this
		NOT = { has_variable = genocide_ended }
    }

	status_desc = je_rwandan_genocide_progress_tt
	immediate = {
		if = {
			limit = { NOT = { has_variable = genocide_progress } }
			set_variable = { name = genocide_progress value = 0 }
		}
	}
	on_monthly_pulse = {
		effect = {
			change_variable = { name = genocide_progress add = 10 }
			if = {
				limit = { var:genocide_progress >= 100 }
				set_variable = genocide_ended
			}
		}
	}


	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		flags = { historical }
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
			custom_tooltip = {
				text = genocide_ended
				has_variable = genocide_ended
			}
			AND = { 
				NOT = { has_modifier = rwandan_genocide }
				custom_description = { text = smd_date_from_1994_5_1 game_date >= 1994.5.1 }
				}
			}
		}
		effect = {
			add_modifier = {
			    name = post_rwandan_genocide_reconstruction
			    months = 8
			}
			add_radicals = { value = -0.05 }
		}
	}
}
