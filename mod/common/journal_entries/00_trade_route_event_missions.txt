je_build_local_arms_industry = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_domestic_policy


	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_state = {
				has_active_building = building_arms_industry
			}
		}
		effect = {
			ig:ig_armed_forces = {
				add_modifier = {
					name = ig_domestic_weapons_manufacturing
					days = short_modifier_time
				}
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			ig:ig_armed_forces = {
				add_modifier = {
					name = ig_unreliable_weapon_imports
					days = short_modifier_time
				}
			}
		}
	}
	timeout = 730

	weight = 200

	transferable = no
}

je_set_up_grain_import = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_foreign_policy


	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			capital = {
				sg:grain = {
					or = {
						import_tariff_level = low_subventions
						import_tariff_level = high_subventions
						import_tariff_level = max_subventions
					}
				}
			}
			capital.market = {
				NOT = {
					mg:grain = { market_goods_pricier > 0.20 }
				}
			}
		}
		effect = {
			scope:target = {
				add_modifier = {
					name = ig_secured_desired_imports
					days = normal_modifier_time
				}
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			scope:target = {
				add_modifier = {
					name = ig_refused_desired_imports
					days = normal_modifier_time
				}
			}
		}
	}
	timeout = 730

	weight = 200

	transferable = no
	should_be_pinned_by_default = yes
}
