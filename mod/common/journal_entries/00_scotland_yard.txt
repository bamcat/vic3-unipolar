je_serial_killer = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	group = je_group_dilemmas

	activation_description = {
		always = yes
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					institution_investment_level = {
						institution = institution_police
						value >= 4
					}
					has_modifier = serial_killer_active_investigation
				}
				random_list = {
					60 = { trigger_event = { id = crime_events.11 } }
					20 = { trigger_event = { id = crime_events.12 } }
					5 = { trigger_event = { id = crime_events.13 } }
					15 = { }
				}
			}
			else_if = {
				limit = {
					OR = {
						institution_investment_level = {
							institution = institution_police
							value >= 4
						}
						AND = {
							institution_investment_level = {
								institution = institution_police
								value >= 2
							}
							has_modifier = serial_killer_active_investigation
						}
					}
				}
				random_list = {
					40 = { trigger_event = { id = crime_events.11 } }
					20 = { trigger_event = { id = crime_events.12 } }
					10 =  { trigger_event = { id = crime_events.13 } }
					50 = { }
				}
			}
			else_if = {
				limit = {
					OR = {
						institution_investment_level = {
							institution = institution_police
							value >= 2
						}
						has_modifier = serial_killer_active_investigation
					}
				}
				random_list = {
					20 = { trigger_event = { id = crime_events.11 } }
					20 = { trigger_event = { id = crime_events.12 } }
					20 = { trigger_event = { id = crime_events.13 } }
					40 = { }
				}
			}
			else = {
				random_list = {
					5 = { trigger_event = { id = crime_events.11 } }
					20 = { trigger_event = { id = crime_events.12 } }
					25 = { trigger_event = { id = crime_events.13 } }
					50 = { }
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = serial_killer_je_tt
				has_variable = serial_killer_caught
			}
		}
		effect = {
			scope:target = {
				remove_modifier = serial_killer_state_mod
				if = {
					limit = {
						has_modifier = modifier_public_worried
					}
					remove_modifier = modifier_public_worried
				}
			}
			remove_modifier = serial_killer_active_investigation
			remove_variable = ongoing_serial_killer_je
			je_serial_killer_cleanup = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = serial_killer_highest_chance
				trigger = {
					institution_investment_level = {
						institution = institution_police
						value >= 4
					}
					has_modifier = serial_killer_active_investigation
				}
			}
			triggered_desc = {
				desc = serial_killer_high_chance
				trigger = {
					OR = {
						institution_investment_level = {
							institution = institution_police
							value >= 4
						}
						AND = {
							institution_investment_level = {
								institution = institution_police
								value >= 2
							}
							has_modifier = serial_killer_active_investigation
						}
					}
				}
			}
			triggered_desc = {
				desc = serial_killer_medium_chance
				trigger = {
					OR = {
						institution_investment_level = {
							institution = institution_police
							value >= 2
						}
						has_modifier = serial_killer_active_investigation
					}
				}
			}
			triggered_desc = {
				desc = serial_killer_low_chance
				trigger = { }
			}
		}
	}
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { exists = scope:target }
				scope:target = {
					OR = {
						NOT = { owner = ROOT }
						NOT = { has_modifier = serial_killer_state_mod }
					}
				}
			}
		}
		effect = {
			scope:target = {
				remove_modifier = serial_killer_state_mod
				if = {
					limit = {
						has_modifier = modifier_public_worried
					}
					remove_modifier = modifier_public_worried
				}
			}
			remove_modifier = serial_killer_active_investigation
			je_serial_killer_cleanup = yes
		}
	}
}
