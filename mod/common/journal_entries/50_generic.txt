smd_je_devolution = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_domestic_policy

	possible = {
		always = yes
	}

	immediate = {
		set_variable = {
			name = smd_devolution_state
			value = scope:target
		}
		scope:target = {
			random_scope_culture = {
				limit = {
					NOT = { is_primary_culture_of = ROOT }
					has_homeland = PREV
					PREV = {
						culture_percent_state = {
							target = PREV
							value >= 0.4
						}
					}
				}
				ROOT = {
					set_variable = {
						name = smd_devolution_culture
						value = PREV
					}
				}
			}
		}
		scope:target = { save_scope_as = smd_devolution_state }
		trigger_event = { id = smd_devolution.1 popup = yes }
	}

	scripted_progress_bar = smd_devolution_bar

	scripted_button = smd_btn_devolution_assembly
	scripted_button = smd_btn_devolution_talks
	scripted_button = smd_btn_devolution_recentralise

	on_monthly_pulse = {
		random_events = {
			100 = 0
			4 = smd_devolution.2
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = smd_je_devolution_success_desc
		trigger = {
			custom_tooltip = {
				text = smd_je_devolution_success_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(smd_devolution_bar)" <= 0
				}
			}
		}
		effect = {
			var:smd_devolution_state ?= { save_scope_as = smd_devolution_state }
			trigger_event = { id = smd_devolution.3 popup = yes }
			remove_variable = smd_devolution_state
			remove_variable = smd_devolution_culture
			remove_variable = smd_recentralised
			set_variable = { name = smd_devolution_cooldown days = 7300 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = smd_je_devolution_failure_desc
		trigger = {
			custom_tooltip = {
				text = smd_je_devolution_failure_tt
				scope:journal_entry ?= {
					"scripted_bar_progress(smd_devolution_bar)" >= 100
				}
			}
		}
		effect = {
			var:smd_devolution_state ?= { save_scope_as = smd_devolution_state }
			trigger_event = { id = smd_devolution.4 popup = yes }
			remove_variable = smd_devolution_state
			remove_variable = smd_devolution_culture
			remove_variable = smd_recentralised
			set_variable = { name = smd_devolution_cooldown days = 7300 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = smd_je_devolution_invalid_desc
		trigger = {
			OR = {
				NOT = { exists = scope:target }
				scope:target = {
					NOT = { owner = ROOT }
				}
			}
		}
		effect = {
			remove_variable = smd_devolution_state
			remove_variable = smd_devolution_culture
			remove_variable = smd_recentralised
			set_variable = { name = smd_devolution_cooldown days = 3650 }
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = smd_je_devolution_timeout_desc
		effect = {
			remove_variable = smd_devolution_state
			remove_variable = smd_devolution_culture
			remove_variable = smd_recentralised
			set_variable = { name = smd_devolution_cooldown days = 3650 }
		}
	}

	timeout = 3650

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}

smd_je_brain_drain = {
	icon = "gfx/interface/icons/invention_icons/academia.dds"

	group = je_group_domestic_policy

	is_shown_when_inactive = {
		has_variable = smd_generic_ready
		NOT = { is_country_type = decentralized }
		NOT = { has_variable = smd_brain_drain_cooldown }
		OR = {
			AND = {
				is_in_power_bloc = yes
				is_power_bloc_leader = no
				power_bloc ?= {
					OR = {
						has_principle = principle:principle_freedom_of_movement_1
						has_principle = principle:principle_freedom_of_movement_2
						has_principle = principle:principle_freedom_of_movement_3
					}
				}
				smd_sol_ratio_to_bloc_leader < 0.75
			}
			AND = {
				has_variable = smd_post_socialist
				any_scope_state = {
					is_incorporated = yes
					population_below_expected_sol >= 0.4
				}
			}
		}
	}

	possible = {
		always = yes
	}

	scripted_progress_bar = smd_braindrain_bar

	scripted_button = smd_btn_braindrain_diaspora
	scripted_button = smd_btn_braindrain_research
