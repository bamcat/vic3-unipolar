je_quebec_happiness = {
    icon = "gfx/interface/icons/event_icons/quebec.dds"
    group = je_group_dilemmas

    is_shown_when_inactive = {
		NOT = { has_variable = quebec_independent }
        c:CAN ?= this
		country_or_subject_owns_entire_state_region = STATE_QUEBEC
		game_date <= 1996.1.1
    }

    possible = {
		NOT = { has_variable = quebec_independent }
        c:CAN ?= this
        country_or_subject_owns_entire_state_region = STATE_QUEBEC
		custom_description = { text = smd_date_until_1996_1_1 game_date <= 1996.1.1 }
    }


	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		flags = { historical }
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = { custom_description = { text = smd_date_from_1996_1_2 game_date >= 1996.1.2 } }
	}
	immediate = {
		if = {
			limit = { NOT = { has_variable = quebec_happiness_var } }
			set_variable = { name = quebec_happiness_var value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = quebec_concessions_used } }
			set_variable = { name = quebec_concessions_used value = 0 }
		}
	}
	on_monthly_pulse = {
		random_events = {
			chance_to_happen = 25
			10 = quebec.3
			10 = quebec.4
			10 = quebec.5
			25 = quebec.6
		}
		effect = {
			if = {
				limit = {
					NOT = { country_or_subject_owns_entire_state_region = STATE_QUEBEC }
				}
				set_variable = quebec_independent
			}
		}
	}

    should_be_pinned_by_default_uninvolved_or_context = yes
    scripted_progress_bar = quebec_happiness_bar
    scripted_button = quebec_concessions_button
    scripted_button = quebec_ignore_button
}
