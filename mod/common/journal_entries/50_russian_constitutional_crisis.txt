	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_wild_nineties

	possible = {
		c:RUS ?= this
	}

	scripted_progress_bar = rus_dvoevlastie_bar

	scripted_button = rus_dvoe_army_button
	scripted_button = rus_dvoe_mvd_button
	scripted_button = rus_dvoe_moscow_button
	scripted_button = rus_dvoe_court_button
	scripted_button = rus_dvoe_regions_button
	scripted_button = rus_dvoe_church_button
	scripted_button = rus_dvoe_zero_option_button
	scripted_button = rus_dvoe_decree_1400_button

	}

		set_variable = { name = rus_inst_army value = 10 }
		set_variable = { name = rus_inst_mvd value = 15 }
		set_variable = { name = rus_inst_moscow value = 30 }
		set_variable = { name = rus_inst_court value = -10 }
		set_variable = { name = rus_inst_regions value = -5 }
		set_variable = { name = rus_inst_church value = 0 }
		trigger_event = { id = rus_events.1 days = 3 popup = yes }
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = rus_inst_army_president_status
				trigger = { var:rus_inst_army >= 20 }
			}
			triggered_desc = {
				desc = rus_inst_army_parliament_status
				trigger = { var:rus_inst_army <= -20 }
			}
			triggered_desc = {
				desc = rus_inst_army_split_status
			}
		}
		first_valid = {
			triggered_desc = {
				desc = rus_inst_mvd_president_status
				trigger = { var:rus_inst_mvd >= 20 }
			}
			triggered_desc = {
				desc = rus_inst_mvd_parliament_status
				trigger = { var:rus_inst_mvd <= -20 }
			}

				desc = rus_inst_mvd_split_status
			}

		first_valid = {
			triggered_desc = {

			}
			triggered_desc = {

				trigger = { var:rus_inst_moscow <= -20 }
			}
			triggered_desc = {
				desc = rus_inst_moscow_split_status
			}
		}
		first_valid = {
			triggered_desc = {
				desc = rus_inst_court_president_status
				trigger = { var:rus_inst_court >= 20 }
			}
			triggered_desc = {
				desc = rus_inst_court_parliament_status

			triggered_desc = {
				desc = rus_inst_court_split_status

		}
		first_valid = {

			triggered_desc = {

	on_monthly_pulse = {
		effect = {
			if = {
				limit = { NOT = { has_variable = rus_decree_1400 } }
				random_list = {
					70 = { }
					5 = { rus_wn_inst_change = { I = army V = -10 } }
					5 = { rus_wn_inst_change = { I = mvd V = -10 } }
					5 = { rus_wn_inst_change = { I = moscow V = -10 } }
					5 = { rus_wn_inst_change = { I = court V = -10 } }
					5 = { rus_wn_inst_change = { I = regions V = -10 } }
					5 = { rus_wn_inst_change = { I = church V = -10 } }
				}
			}
			if = {
				limit = {
					game_date >= 1992.12.9
					NOT = { has_variable = rus_dvoe_congress_1992 }
				}
				set_variable = rus_dvoe_congress_1992
				trigger_event = { id = rus_events.2 popup = yes }
			}
			if = {
				limit = {
					game_date >= 1993.3.20
					NOT = { has_variable = rus_dvoe_impeachment_vote }
					NOT = { has_variable = rus_1993_result }
				}
				set_variable = rus_dvoe_impeachment_vote
				trigger_event = { id = rus_events.3 popup = yes }
			}
			if = {
				limit = {
					game_date >= 1993.4.25
					NOT = { has_variable = rus_1993_referendum_held }
					NOT = { has_variable = rus_1993_result }
				}
				set_variable = rus_1993_referendum_held
				trigger_event = { id = rus_events.4 popup = yes }
			}
			if = {
				limit = {
					game_date >= 1993.6.5
					NOT = { has_variable = rus_dvoe_conference }
					NOT = { has_variable = rus_1993_result }
				}
				set_variable = rus_dvoe_conference
				trigger_event = { id = rus_events.5 popup = yes }
			}
			if = {
				limit = {
					game_date >= 1993.12.1
					NOT = { has_variable = rus_1993_result }
					NOT = { has_variable = rus_decree_1400 }
					NOT = { has_variable = rus_dvoe_year_end }
				}
				set_variable = rus_dvoe_year_end
				trigger_event = { id = rus_events.10 popup = yes }
			}
		}
	}
