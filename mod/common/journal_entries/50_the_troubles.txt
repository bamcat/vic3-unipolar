je_troubles_in_northern_ireland = {
	icon = "gfx/interface/icons/generic_icons/battalions.dds"
	group = je_group_dilemmas
	should_be_pinned_by_default_uninvolved_or_context = yes

	is_shown_when_inactive = {
		c:GBR ?= this
		NOT = { has_global_variable = good_friday_agreement }
		any_scope_state = { state_region = s:STATE_ULSTER }
	}

	possible = {
		c:GBR ?= this
		NOT = { has_global_variable = good_friday_agreement }
		any_scope_state = { state_region = s:STATE_ULSTER }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		flags = { historical }
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = good_friday_agreement_tt
				has_global_variable = good_friday_agreement
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { any_scope_state = { state_region = s:STATE_ULSTER } }
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = smd_ni_withdrawal_cooldown }
					je:je_troubles_in_northern_ireland ?= {
						"scripted_bar_progress(troubles_troop_strength_bar)" <= 5
					}
				}
				trigger_event = { id = ni_peace_events.5 popup = yes }
			}
		}
		random_events = {
			chance_to_happen = 50
			25 = troubles_events.1
			25 = troubles_events.2
			20 = troubles_events.3
			15 = troubles_events.4
			15 = troubles_events.5
		}
	}

	scripted_button = je_troubles_increase_troop_button
	scripted_progress_bar = troubles_troop_strength_bar
}
