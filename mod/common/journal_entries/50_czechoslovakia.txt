je_velvet_divorce = {
	icon = "gfx/interface/icons/event_icons/czechoslovakia.dds"
	group = je_group_dilemmas
 
	should_be_pinned_by_default_uninvolved_or_context = yes
 
	is_shown_when_inactive = {
		c:CZH ?= this
		game_date >= 1992.1.1
		NOT = { has_variable = velvet_divorce_resolved }
	}
 
	possible = {
		c:CZH ?= this
		custom_description = { text = smd_date_from_1992_1_1 game_date >= 1992.1.1 }
		NOT = { has_variable = velvet_divorce_resolved }
	}
 
	immediate = {
		set_variable = { name = separatist_pressure value = 6 }
		if = {
			limit = { NOT = { has_variable = federal_cohesion } }
			set_variable = { name = federal_cohesion value = 0 }
		}
	}
 
	scripted_progress_bar = federal_cohesion_bar
	scripted_progress_bar = separatist_pressure_bar
 
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_velvet_divorce_critical_tt
				trigger = { var:separatist_pressure >= 75 }
			}
			triggered_desc = {
				desc = je_velvet_divorce_tense_tt
				trigger = { var:separatist_pressure >= 50 }
			}
			triggered_desc = {
				desc = je_velvet_divorce_stable_tt
			}
		}
	}
 
	on_monthly_pulse = {
		effect = {
			change_variable = { name = separatist_pressure add = 3 }
 
			change_variable = { name = federal_cohesion add = -1 }
 
			if = {
				limit = { legitimacy >= 70 }
				change_variable = { name = separatist_pressure add = -1 }
			}
 
			if = {
				limit = {
					NOT = { authority >= 100 }
				}
				change_variable = { name = separatist_pressure add = 1 }
			}
		}
 
		random_events = {
			3 = velvet_divorce.2
			3 = velvet_divorce.3
			2 = velvet_divorce.4
			2 = velvet_divorce.5
		}
	}
 
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			je:je_velvet_divorce = {
			"scripted_bar_progress(federal_cohesion_bar)" >= 96
					}
		}
		effect = {
			set_variable = velvet_divorce_resolved
			trigger_event = { id = velvet_divorce.10 popup = yes }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		flags = { historical }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			je:je_velvet_divorce = {
			"scripted_bar_progress(separatist_pressure_bar)" >= 96
					}
		}
		effect = {
			set_variable = velvet_divorce_resolved
			trigger_event = { id = velvet_divorce.15 popup = yes }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { NOT = { c:CZH ?= this } }
	}

	weight = 5000
	transferable = no
	can_revolution_inherit = no
}
