# ==========================================
# 2008 FINANCIAL CRISIS

je_subprime_mortgage_crisis = {
	icon = "gfx/interface/icons/event_icons/subprime_mortgage_crisis.dds"

	group = je_group_dilemmas

	should_be_pinned_by_default_uninvolved_or_context = yes
	
	is_shown_when_inactive = {
		c:USA ?= this
		any_scope_building = {
			is_building_type = building_financial_district
			has_active_production_method = pm_financial_district_subprime_mortgages
		}
	}

	possible = {
		c:USA ?= this
		any_scope_building = {
			is_building_type = building_financial_district
			has_active_production_method = pm_financial_district_subprime_mortgages
		}
	}

	should_be_involved = {
		c:USA ?= this
	}

outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			market_capital.market = {
				mg:securities = {
					market_goods_cheaper > 0.35
				}
			}
		}
		effect = {}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout historical }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_journal_entry = { type = je_great_recession }
			every_country = {
				limit = {
					NOT = { c:USA ?= this }
					exists = c:USA
					market_capital.market = {
						market_trade = {
							target = c:USA.market_capital.market
							value > 5000
						}
					}
				}
				add_journal_entry = { type = je_great_recession }
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_building = {
					is_building_type = building_financial_district
					has_active_production_method = pm_financial_district_subprime_mortgages
				}
			}
		}
	}

	# 8 months to correct before the crash

	transferable = no
	can_revolution_inherit = no
	weight = 8000
	timeout = 240
}

	outcome = {

	}
}
