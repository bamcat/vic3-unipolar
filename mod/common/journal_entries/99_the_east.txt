	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/objectives/the_east.dds"

	group = je_group_objectives

		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				has_law = law_type:law_interventionism

			}
		}
	}
	is_progressing = {
		OR = {
			has_law = law_type:law_interventionism
			has_law = law_type:law_command_economy
		}
	}

	weight = 3000
	transferable = yes
}

	icon = "gfx/interface/icons/objectives/the_east.dds"

	group = je_group_objectives

		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_country = {
				hidden_trigger = { NOT = { THIS = ROOT } }
				country_rank >= rank_value:major_power
				any_scope_treaty = {
					binds = root
					any_scope_article = {
						has_type = alliance
					}
				}
				count >= 2
			}
		}
	}
	is_progressing = {
		any_country = {
			hidden_trigger = { NOT = { THIS = ROOT } }
			country_rank >= rank_value:major_power
			any_scope_treaty = {
				binds = root
				any_scope_article = {
					has_type = alliance
				}
			}
			count >= 1
		}
	}

	weight = 3002
	transferable = yes
}
