# ── SG 1: REGIONAL ECONOMY ───────────────
# dependency.

je_rising_economy = {
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/objectives/na.dds"

	group = je_group_objectives

outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			gdp > {
				value = global_gdp
				multiply = 0.05
			}
			NOT = {
				market_capital.market = {
					market_trade_reliance = {
						target = c:USA.market_capital.market
						value > 0.40
					}
				}
			}
		}
	}
	is_progressing = {
		gdp > {
			value = global_gdp
			multiply = 0.02
		}
	}

	weight = 4000
	transferable = yes
}

# ── SG 5: THE THIRD BLOC ─────────────────

je_rising_endgame = {
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/objectives/na.dds"

	group = je_group_objectives

outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_power_bloc_leader = yes

				num_power_bloc_members >= 8
			}
		}
	}
	is_progressing = {
		power_bloc ?= {
			num_power_bloc_members >= 4
		}
	}

	weight = 4004
	transferable = yes
}
