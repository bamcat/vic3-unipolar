je_uranium_enrichment = {
	icon = "gfx/interface/icons/event_icons/nuclear_bomb.dds"
	group = je_group_domestic_policy
 
	should_be_pinned_by_default_uninvolved_or_context = yes
 
	is_shown_when_inactive = {
		has_variable = nuclear_enrichment_active
		NOT = { has_variable = nuclear_capable }
	}
 
	possible = {
		has_variable = nuclear_enrichment_active
	}
 
	on_monthly_pulse = {
		effect = {
			change_variable = { name = uranium_enrichment_progress add = 2 }
 
			if = {
				limit = {
					country_has_building_type_levels = {
						target = bt:building_nuclear_plant
						value >= 4
					}
				}
				change_variable = { name = uranium_enrichment_progress add = 1 }
			}
			if = {
				limit = {
					country_has_building_type_levels = {
						target = bt:building_nuclear_plant
						value >= 6
					}
				}
				change_variable = { name = uranium_enrichment_progress add = 1 }
			}
 
			if = {
				limit = { is_at_war = yes }
				change_variable = { name = uranium_enrichment_progress add = -1 }
			}
 
			if = {
				limit = { var:uranium_enrichment_progress >= 100 }
				set_variable = { name = uranium_enrichment_progress value = 100 }
			}
		}
 
		random_events = {
			10 = 0
			2 = nuclear_events.2
		}
	}



	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = { var:uranium_enrichment_progress >= 100 }
		effect = {
			remove_variable = nuclear_enrichment_active
			set_variable = nuclear_capable
			trigger_event = { id = nuclear_events.1 popup = yes }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { NOT = { has_variable = nuclear_enrichment_active } }
	}	weight = 8000
	transferable = no
	can_revolution_inherit = no
}

je_nuclear_production = {
	icon = "gfx/interface/icons/event_icons/nuclear_bomb.dds"
	group = je_group_domestic_policy
 
	should_be_pinned_by_default_uninvolved_or_context = yes
 
	is_shown_when_inactive = {
		has_variable = nuclear_production_active
	}
 
	possible = {
		has_variable = nuclear_production_active
	}
 
	# scripted_progress_bar = nuclear_production_progress_bar
 
	on_monthly_pulse = {
		effect = {
			change_variable = { name = nuclear_production_progress add = 5 }
 
			if = {
				limit = {
					any_scope_state = {
						is_incorporated = yes
						has_building = building_nuclear_silo
						count >= 2
					}
				}
				change_variable = { name = nuclear_production_progress add = 2 }
			}
 
			if = {
				limit = { var:nuclear_production_progress >= 100 }
				set_variable = { name = nuclear_production_progress value = 100 }
			}
		}
	}



	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = { var:nuclear_production_progress >= 100 }
		effect = {
			remove_variable = nuclear_production_active
			if = {
				limit = { NOT = { has_variable = nukes } }
				set_variable = { name = nukes value = 1 }
			}
			else = {
				change_variable = { name = nukes add = 1 }
			}
			set_variable = { name = nuclear_production_progress value = 0 }
			if = {
				limit = { NOT = { has_variable = first_warhead_produced } }
				set_variable = first_warhead_produced
				add_modifier = {
					name = nuclear_warhead_produced_modifier
					months = 24
				}
			}
			post_notification = nuclear_warhead_produced

			# production run automatically.
			if = {
				limit = {
					has_variable = nuclear_production_queue
					var:nuclear_production_queue > 0
					any_scope_state = {
						is_incorporated = yes
						has_building = building_nuclear_silo
					}
				}
				change_variable = { name = nuclear_production_queue add = -1 }
				set_variable = nuclear_production_active
				set_variable = { name = nuclear_production_progress value = 0 }
				trigger_event = { id = um_journal_events.2 days = 1 }
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { NOT = { has_variable = nuclear_production_active } }
	}	weight = 8001
	transferable = no
	can_revolution_inherit = no
}
