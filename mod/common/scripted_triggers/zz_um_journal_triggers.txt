	}

arab_spring_islamist_victory_trigger = {
	any_interest_group = {
		is_in_government = yes
		is_powerful = yes
		leader = {
			OR = {
				has_ideology = ideology:ideology_theocrat_leader
				has_ideology = ideology:ideology_jihadist_leader
			}
		}
	}
	je:je_arab_winter ?= {
		"scripted_bar_progress(arab_spring_progress_bar)" >= 50
		"scripted_bar_progress(arab_winter_crackdown_progress_bar)" < 100
	}
}
