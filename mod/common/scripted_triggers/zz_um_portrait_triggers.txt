um_cleric_character_trigger = {
	exists = scope:character
	scope:character = {
		is_adult = yes
		is_female = no
		religion = {
			has_discrimination_trait = heritage_islamic
		}
		NOR = {
			has_role_of_type = general
			has_role_of_type = admiral
		}
		NOT = {
			culture = {
				has_discrimination_trait = heritage_anatolian
			}
		}
		OR = {
			AND = {
				is_interest_group_type = ig_devout
				culture = {
					OR = {
						has_discrimination_trait_group = heritage_group_middle_eastern
						has_discrimination_trait_group = heritage_group_central_asian
					}
				}
			}
			AND = {
				is_interest_group_type = ig_devout
				home_country ?= {
					has_law_or_variant = law_type:law_theocracy
				}
			}
			AND = {
				has_ideology = ideology:ideology_theocrat_leader
				is_ruler_of_own_country = yes
				home_country ?= {
					has_law_or_variant = law_type:law_theocracy
				}
			}
		}
	}
}

	exists = scope:character
	scope:character = {
		is_adult = yes
		OR = {
			has_role_of_type = general
			has_role_of_type = admiral
			AND = {
				is_interest_group_type = ig_armed_forces
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
					is_ruler_of_own_country = yes
				}
			}
			AND = {
				is_ruler_of_own_country = yes
				is_historical = yes
				home_country ?= {
					c:LBY ?= this
				}
			}
		}
	}
	um_cleric_character_trigger = no

}
