rus_wn_has_character = {
	custom_description = {
		text = rus_wn_has_character_$T$
		any_scope_character = {
			has_template = $T$
			is_character_alive = yes
		}
	}
}

rus_wn_is_pm = {
	custom_description = {
		text = rus_wn_is_pm_$T$
		any_scope_character = {
			has_template = $T$
			has_role = character_role_prime_minister
		}
	}
}

rus_wn_can_be_pm = {
	custom_description = {
		text = rus_wn_can_be_pm_$T$
		any_scope_character = {
			has_template = $T$
			is_character_alive = yes
			NOT = { has_role = character_role_prime_minister }
			is_ruler_of_own_country = no
		}
	}
}

rus_wn_institutions_backing_president = {
	calc_true_if = {
		amount >= $N$
		var:rus_inst_army >= 20
		var:rus_inst_mvd >= 20
		var:rus_inst_moscow >= 20
		var:rus_inst_court >= 20
		var:rus_inst_regions >= 20
		var:rus_inst_church >= 20
	}
}

rus_wn_chechen_war_possible = {
	custom_description = {
		text = rus_wn_chechen_war_possible
		NOT = { has_variable = chechen_war_started }
		NOT = { has_global_variable = chechen_war_started }
		NOT = { has_variable = rus_l90_ichkeria_started }
		exists = c:CCH
		c:CCH ?= {
			any_scope_state = { is_incorporated = yes }
			NOT = { has_war_with = c:RUS }
			is_subject = no
		}
	}
}
