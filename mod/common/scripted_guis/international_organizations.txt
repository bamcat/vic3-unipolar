io_join_csto = {
	scope = country
	is_shown = { has_global_variable = csto_founded }
	is_valid = {
		NOT = { c:RUS ?= this }
		NOT = { is_target_in_global_variable_list = { name = io_csto_members target = ROOT } }
		custom_description = {
			text = rus_l90_csto_in_cis
			is_target_in_global_variable_list = { name = io_cis_members target = ROOT }
		}
		NOT = { is_target_in_global_variable_list = { name = io_nato_members target = ROOT } }
		custom_description = {
			text = rus_l90_csto_russia_friendly
			c:RUS ?= {
				relations:root >= relations_threshold:cordial
				NOT = { has_war_with = root }
			}
		}
		NOT = { has_variable = rus_l90_csto_left_recently }
	}
	effect = {
		custom_tooltip = rus_l90_join_csto_tt
		rus_l90_csto_add_member = yes
	}
}

io_leave_csto = {
	scope = country
	is_shown = {
		has_global_variable = csto_founded
		is_target_in_global_variable_list = { name = io_csto_members target = ROOT }
	}
	is_valid = {
		NOT = { c:RUS ?= this }
	}
	effect = {
		custom_tooltip = rus_l90_leave_csto_tt
		rus_l90_csto_remove_member = yes
		hidden_effect = {
			set_variable = { name = rus_l90_csto_left_recently years = 5 }
		}
		post_notification = io_left_org_notification
	}
}
