# Diplomatic play: Great Power vs Great Power is always covered. 

news_events_world_diplomatic_play_started = {
    trigger = {
        initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        target = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        NOT = { has_variable = news_world_dp_reported }
        NOT = {
            OR = {
                AND = { has_global_variable = 911_usa initiator = { c:USA ?= this } target = { OR = { c:AFG ?= this c:IRQ ?= this } } }
                AND = { has_global_variable = 911_usa target = { c:USA ?= this } initiator = { OR = { c:AFG ?= this c:IRQ ?= this } } }
            }
        }
        OR = {
            AND = { initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power } target = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power } }
            AND = { initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power OR = { infamy < 1 has_variable = news_world_was_zero_infamy } } target = { AND = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } OR = { AND = { country_rank < rank_value:great_power NOT = { is_country_type = unrecognized } } country_rank = rank_value:unrecognized_major_power country_rank = rank_value:unrecognized_regional_power } } } }
            AND = { target = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power OR = { infamy < 1 has_variable = news_world_was_zero_infamy } } initiator = { AND = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } OR = { AND = { country_rank < rank_value:great_power NOT = { is_country_type = unrecognized } } country_rank = rank_value:unrecognized_major_power country_rank = rank_value:unrecognized_regional_power } } } }
        }
    }
    effect = {
        save_scope_as = news_diplomatic_play
        set_variable = news_world_dp_reported
        initiator = { save_scope_as = news_country_a }
        target = { save_scope_as = news_country_b }
        initiator = { capital = { save_scope_as = news_event_state } }
        if = {
            limit = { initiator = { country_rank = rank_value:great_power } target = { country_rank = rank_value:great_power } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.1 } }
        }
        else = {
            if = { limit = { initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power OR = { infamy < 1 has_variable = news_world_was_zero_infamy } } } initiator = { save_scope_as = news_clean_gp } target = { save_scope_as = news_other_country } }
            else = { target = { save_scope_as = news_clean_gp } initiator = { save_scope_as = news_other_country } }
            set_variable = news_world_clean_gp_play
            set_variable = { name = news_world_clean_gp_country value = scope:news_clean_gp }
            if = {
                limit = { scope:news_other_country = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } OR = { NOT = { is_country_type = unrecognized } country_rank = rank_value:unrecognized_major_power } } }
                every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.2 } }
            }
        }
    }
}

news_events_world_war_started = {
    trigger = {
        initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        target = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        NOT = { has_variable = news_world_war_reported }
        NOT = {
            OR = {
                AND = { initiator = { has_variable = opium_wars_aggressor } target = { has_variable = opium_wars_target } }
                AND = { target = { has_variable = opium_wars_aggressor } initiator = { has_variable = opium_wars_target } }
                AND = { OR = { AND = { initiator = { c:ISR ?= this } target = { c:PAL ?= this } } AND = { initiator = { c:PAL ?= this } target = { c:ISR ?= this } } } }
                AND = { has_global_variable = 911_usa initiator = { c:USA ?= this } target = { OR = { c:AFG ?= this c:IRQ ?= this } } }
                AND = { has_global_variable = 911_usa target = { c:USA ?= this } initiator = { OR = { c:AFG ?= this c:IRQ ?= this } } }
            }
        }
        OR = {
            AND = { initiator = { country_rank >= rank_value:major_power } target = { country_rank >= rank_value:major_power } }
            has_variable = news_world_clean_gp_play
            AND = { initiator = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power OR = { infamy < 1 has_variable = news_world_was_zero_infamy } } target = { OR = { AND = { country_rank < rank_value:major_power NOT = { is_country_type = unrecognized } } country_rank = rank_value:unrecognized_major_power } } }
            AND = { target = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } country_rank = rank_value:great_power OR = { infamy < 1 has_variable = news_world_was_zero_infamy } } initiator = { OR = { AND = { country_rank < rank_value:major_power NOT = { is_country_type = unrecognized } } country_rank = rank_value:unrecognized_major_power } } }
        }
    }
    effect = {
        save_scope_as = news_diplomatic_play
        set_variable = news_world_war_reported
        initiator = { save_scope_as = news_country_a capital = { save_scope_as = news_event_state } }
        target = { save_scope_as = news_country_b }
        if = {
            limit = { initiator = { country_rank >= rank_value:major_power } target = { country_rank >= rank_value:major_power } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.3 } }
        }
        else = {
            if = {
                limit = { exists = var:news_world_clean_gp_country }
                var:news_world_clean_gp_country = { save_scope_as = news_clean_gp }
                if = { limit = { target = { NOT = { this = scope:news_clean_gp } } } target = { save_scope_as = news_other_country } }
                else = { initiator = { save_scope_as = news_other_country } }
            }
            else_if = { limit = { initiator = { country_rank = rank_value:great_power } } initiator = { save_scope_as = news_clean_gp } target = { save_scope_as = news_other_country } }
            else = { target = { save_scope_as = news_clean_gp } initiator = { save_scope_as = news_other_country } }
            if = {
                limit = { scope:news_other_country = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } OR = { NOT = { is_country_type = unrecognized } country_rank = rank_value:unrecognized_major_power } } }
                every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.4 } }
            }
        }
    }
}

# A zero-infamy Great Power can become involved after the play has started via a sway. If the US invaded cuba after 20 years of doing fuck all, it should be covered
news_events_world_country_swayed = {
    trigger = { NOT = { has_variable = news_world_dp_reported } }
    effect = {
        save_scope_as = news_diplomatic_play
        initiator = { save_scope_as = news_play_initiator }
        target = { save_scope_as = news_play_target }
        if = {
            limit = {
                any_country = {
                    country_rank = rank_value:great_power
                    OR = { infamy < 1 has_variable = news_world_was_zero_infamy }
                    is_diplomatic_play_participant_with = scope:news_play_initiator
                    is_diplomatic_play_participant_with = scope:news_play_target
                }
            }
            random_country = {
                limit = {
                    country_rank = rank_value:great_power
                    OR = { infamy < 1 has_variable = news_world_was_zero_infamy }
                    is_diplomatic_play_participant_with = scope:news_play_initiator
                    is_diplomatic_play_participant_with = scope:news_play_target
                }
                save_scope_as = news_clean_gp
                capital = { save_scope_as = news_event_state }
            }
            target = { save_scope_as = news_other_country }
            set_variable = news_world_dp_reported
            set_variable = news_world_clean_gp_play
            set_variable = { name = news_world_clean_gp_country value = scope:news_clean_gp }
            if = {
                limit = { scope:news_other_country = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } OR = { NOT = { is_country_type = unrecognized } country_rank = rank_value:unrecognized_major_power } } }
                every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.2 } }
            }
        }
    }
}


# DIPLOMATIC PLAY BACKDOWN
# Root = Diplomatic Play. Vanilla supplies scope:actor as the country backing down
# and scope:target as its opponent.
news_events_world_diplo_play_back_down = {
    trigger = {
        exists = scope:actor
        scope:actor = {
            OR = {
                country_rank = rank_value:great_power
                country_rank = rank_value:major_power
            }
            global_country_ranking <= 125
            NOT = { is_revolutionary = yes }
            news_is_reportable_country = yes
        }
        NOT = { scope:target ?= { c:EUR ?= this } }
    }
    effect = {
        save_scope_as = news_diplomatic_play
        scope:actor = {
            save_scope_as = news_backdown_country
            capital = { save_scope_as = news_event_state }
        }
        scope:target ?= { save_scope_as = news_backdown_target }
        every_country = {
            limit = { is_player = yes }
            trigger_event = { id = news_events_world.32 }
        }
    }
}

news_events_world_character_death = {
    trigger = {
        is_ruler_of_own_country = yes
        exists = owner
        owner = { country_rank >= rank_value:major_power news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        NOT = { has_variable = abraham_lincoln }
    }
    effect = {
        save_scope_as = news_character
        owner = { save_scope_as = news_source_country capital = { save_scope_as = news_event_state } }
        if = {
            limit = { has_variable = assassination_ruler_var owner = { NOT = { has_variable = news_world_assassination_cooldown } } }
            owner = { set_variable = { name = news_world_assassination_cooldown months = 6 } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.6 popup = yes } }
        }
        else_if = {
            limit = { owner = { country_rank = rank_value:great_power NOT = { has_variable = news_world_leader_death_cooldown } } }
            owner = { set_variable = { name = news_world_leader_death_cooldown months = 6 } set_variable = { name = news_world_recent_ruler_death months = 2 } }
            if = { limit = { has_variable = news_world_recent_ruler_death } save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.7 popup = yes } }
        }
        else = {
            owner = { set_variable = { name = news_world_recent_ruler_death months = 2 } }
            if = { limit = { has_variable = news_world_recent_ruler_death } save_scope_as = news_error_log_silencer } # Silence localization-only variable warning.
        }
    }
}

news_events_world_ruler_selected = {
    trigger = { exists = scope:previous exists = owner
        owner = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } } }
    effect = {
        # A living predecessor in a Great Power is a removal, except elections,
        # coups and active revolutions which have their own newspaper categories.
        if = {
            limit = {
                scope:previous = { is_character_alive = yes }
                owner = {
                    country_rank = rank_value:great_power
                    NOT = { has_variable = news_world_election_transition_window }
                    NOT = { has_variable = coup_aftermath_var }
                    NOT = { has_modifier = modifier_government_recently_couped }
                    NOT = { has_journal_entry = je_ip4_coup }
                    has_revolution = no
                    NOT = { has_variable = news_world_leader_removal_cooldown }
                }
            }
            scope:previous = { save_scope_as = news_previous_ruler }
            save_scope_as = news_new_ruler
            owner = { save_scope_as = news_source_country capital = { save_scope_as = news_event_state } set_variable = { name = news_world_leader_removal_cooldown months = 6 } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.8 } }
        }
        # A dead predecessor produces a succession headline in Great/Major Powers.
        # Suppress the American Civil War's bespoke presidential chain.
        else_if = {
            limit = {
                scope:previous = { is_character_alive = no NOT = { has_variable = abraham_lincoln } }
                owner = {
                    country_rank >= rank_value:major_power
                    NOT = { has_variable = news_world_succession_cooldown }
                }
            }
            save_scope_as = news_new_ruler
            owner = { save_scope_as = news_source_country capital = { save_scope_as = news_event_state } set_variable = { name = news_world_succession_cooldown months = 6 } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.16 days = 7 } }
        }
    }
}

news_events_world_treaty_entered_force = {
    trigger = { first_country = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        second_country = { news_is_reportable_country = yes NOT = { is_revolutionary = yes } }
        OR = { any_scope_article = { has_type = state_transfer } any_scope_article = { has_type = alliance } } }
    effect = {
        save_scope_as = news_treaty
        # One territorial headline per treaty entry into force, even if several
        # state-transfer articles are bundled in the same settlement.
        if = {
            limit = { any_scope_article = { has_type = state_transfer } NOT = { has_variable = news_world_state_transfer_reported } }
            set_variable = news_world_state_transfer_reported
            random_scope_article = {
                limit = { has_type = state_transfer }
                input_state = { save_scope_as = news_event_state }
                source_country = { save_scope_as = news_state_giver }
                target_country = { save_scope_as = news_state_taker }
            }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.13 } }
        }
        if = {
            limit = {
                any_scope_article = { has_type = alliance }
                NOT = { has_variable = news_world_alliance_reported }
                OR = { first_country = { country_rank = rank_value:great_power NOT = { has_variable = news_world_alliance_cooldown } } second_country = { country_rank = rank_value:great_power NOT = { has_variable = news_world_alliance_cooldown } } }
            }
            set_variable = news_world_alliance_reported
            first_country = { save_scope_as = news_country_a }
            second_country = { save_scope_as = news_country_b }
            if = { limit = { first_country = { country_rank = rank_value:great_power } } first_country = { capital = { save_scope_as = news_event_state } } }
            else = { second_country = { capital = { save_scope_as = news_event_state } } }
            first_country = { if = { limit = { country_rank = rank_value:great_power } set_variable = { name = news_world_alliance_cooldown months = 6 } } }
            second_country = { if = { limit = { country_rank = rank_value:great_power } set_variable = { name = news_world_alliance_cooldown months = 6 } } }
            every_country = { limit = { is_player = yes } trigger_event = { id = news_events_world.17 } }
        }
    }
}

# Major government changes in Great Powers. Routine switches between presidential
# and parliamentary republics are not enough by themselves; transitions to/from
# monarchy, theocracy, council republic, corporate state, autocracy, single-party
# state or anarchy are treated as systemic regime changes.
news_events_world_law_activated = {
    trigger = {
        owner = {
            news_is_reportable_country = yes NOT = { is_revolutionary = yes }
            country_rank = rank_value:great_power
            NOT = { has_variable = news_world_regime_change_cooldown }
            NOT = { has_variable = coup_aftermath_var }
            NOT = { has_modifier = modifier_government_recently_couped }
            NOT = { has_journal_entry = je_ip4_coup }
            has_revolution = no
        }
        OR = {
            root.type ?= law_type:law_monarchy
            root.type ?= law_type:law_theocracy
            root.type ?= law_type:law_council_republic
            root.type ?= law_type:law_corporate_state

        scope:actor = { save_scope_as = news_country_a }
        scope:target_country = { save_scope_as = news_country_b }
        if = {
            limit = { exists = scope:first_state }
            scope:first_state = { save_scope_as = news_trade_state_a save_scope_as = news_event_state }
        }
        if = {
            limit = { exists = scope:second_state }
            scope:second_state = {
                save_scope_as = news_trade_state_b
                if = { limit = { NOT = { exists = scope:news_event_state } } save_scope_as = news_event_state }
            }
        }
        every_country = {
            limit = { is_player = yes }
            trigger_event = { id = news_events_world.30 }
        }
    }
}

# WORKERS' STRIKE
news_events_world_worker_strike_monthly = {
    trigger = { always = yes }
    effect = {
        every_country = {
            if = {
                limit = {
                    news_is_reportable_country = yes NOT = { is_revolutionary = yes }
                    country_rank = rank_value:great_power
                    OR = {
                        has_journal_entry = je_strike
                        has_modifier = modifier_nationwide_strikes
                        any_scope_state = { has_modifier = modifier_movement_worker_strikes }
                    }
                }
                if = {
                    limit = { NOT = { has_variable = news_world_workers_strike_active } }
                    set_variable = news_world_workers_strike_active
                    save_scope_as = news_source_country

                    # Prefer the largest state that is explicitly carrying a
                    # strike modifier. The dedicated Strike JE adds industrial,
                    # mining or railway modifiers after the government's first
                    # response; movement strikes have their own state modifier.
                    ordered_scope_state = {
                        limit = {
                            OR = {
                                has_modifier = modifier_movement_worker_strikes
                                has_modifier = industrial_strike_modifier
                                has_modifier = mining_strike_modifier
                                has_modifier = railway_strike_modifier
                            }
                        }
                        order_by = state_population
                        position = 0
                        check_range_bounds = no
                        save_scope_as = news_event_state
                    }
                    if = {
                        limit = { NOT = { exists = scope:news_event_state } }
                        capital = { save_scope_as = news_event_state }
                    }

                    every_country = {
                        limit = { is_player = yes }
                        trigger_event = { id = news_events_world.31 }
                    }
                }
            }
            else_if = {
                limit = { has_variable = news_world_workers_strike_active }
                remove_variable = news_world_workers_strike_active
            }
        }
    }
}
