ai_strategy_smd_formation_ambitions = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"

	type = diplomatic

	will_form_power_bloc = {
		country_rank >= rank_value:great_power
	}

	undesirable_infamy_level = {
		value = 50
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = {
		value = 0.75
	}

	wargoal_maneuvers_fraction = {
		value = 0.15
	}

	diplomatic_play_neutrality = {
		value = 25
	}

	diplomatic_play_boldness = {
		value = 25
	}

	recklessness = {
		value = 0.25
	}

	aggression = {
		value = 0.5

		if = {
			limit = {
				scope:target_country = {
					smd_ai_formation_unprotected_target_country = yes
				}
			}
			add = 3
		}
		else_if = {
			limit = {
				scope:target_country = {
					smd_ai_formation_partner_protected = yes
					any_scope_state = {
						smd_ai_formation_target_state = yes
					}
				}
			}
			multiply = 0.1
		}

		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.75
		}

		multiply = {
			value = "ai_ideological_opinion(scope:target_country)"
			multiply = -0.1
			add = 1
			max = 2.0
			min = 0.5
		}
	}

	building_group_weights = {
		bg_army = 1.5
	}

	secret_goal_weights = {
		conquer = 1.5
		dominate = 1.5
	}

	wargoal_weights = {
		conquer_state = 1.5
		annex_country = 1.5
	}

	state_value = {
		if = {
			limit = {
				scope:target_state = {
					smd_ai_formation_target_state = yes
					owner = {
						smd_ai_formation_partner_protected = no
					}
				}
				can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
			}
			add = {
				value = 100
				desc = "STATE_VALUE_SMD_FORMATION_AMBITION"
			}
		}
	}

	strategic_region_scores = {
		if = {
			limit = {
				scope:region = {
					any_scope_state = {
						smd_ai_formation_target_state = yes
						owner = {
							smd_ai_formation_partner_protected = no
						}
					}
				}
			}
			add = 250
		}
	}

	wargoal_scores = {
		conquer_state = {
			if = {
				limit = {
					scope:target_state = {
						smd_ai_formation_target_state = yes
						owner = {
							smd_ai_formation_partner_protected = no
						}
					}
				}
				add = 200
			}
		}
		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						smd_ai_formation_unprotected_target_country = yes
						NOT = {
							any_scope_state = {
								smd_ai_formation_target_state = no
							}
						}
					}
				}
				add = 300
			}
		}
		make_subject = {
			if = {
				limit = {
					scope:target_country = {
						smd_ai_formation_unprotected_target_country = yes
						country_rank < root.country_rank
					}
				}
				add = 150
			}
		}
	}

	secret_goal_scores = {
		conquer = {
			trigger = {
				scope:target_country = {
					smd_ai_formation_unprotected_target_country = yes
				}
				can_reach_target_country_to_conquer = { TARGET_COUNTRY = scope:target_country }
			}
			score = {
				add = 100

				if = {
					limit = {
						has_strategic_adjacency = scope:target_country
					}
					add = 50
				}
			}
		}
		dominate = {
			trigger = {
				scope:target_country = {
					smd_ai_formation_unprotected_target_country = yes
					country_rank < root.country_rank
				}
				can_reach_target_country_to_subjugate = { TARGET_COUNTRY = scope:target_country }
			}
			score = {
				add = 75
			}
		}
		befriend = {
			trigger = {
				scope:target_country = {
					NOT = { this = root }
					smd_ai_formation_partner_protected = yes
					any_scope_state = {
						smd_ai_formation_target_state = yes
					}
				}
			}
			score = {
				add = 100
			}
		}
	}

	treaty_category_scores = {
		expansion = {
			value = 4
		}

		military = {
			value = 3
		}
	}

	possible = {
		is_subject = no
		any_country = {
			smd_ai_formation_unprotected_target_country = yes
		}
	}

	weight = {
		value = 40

		if = {
			limit = { ig:ig_armed_forces ?= { is_powerful = yes } }
			multiply = 1.5
		}
	}
}
