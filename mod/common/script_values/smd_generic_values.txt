smd_sol_ratio_to_bloc_leader = {
	value = 1
	if = {
		limit = {
			power_bloc ?= {
				power_bloc_leader ?= {
					average_sol > 0
				}
			}
		}
		value = average_sol
		divide = power_bloc.power_bloc_leader.average_sol
	}
}
