democratic_party = {
	color = { 0 100 200 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_us_democrats
				trigger = {
					always = yes
				}
			}
		}
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/us_democrats.dds"
	}

	valid_for_country = {
		exists = c:USA
		THIS = c:USA
	}

	available_for_interest_group = {
		is_marginal = no
		NOT = {
			leader ?= {
				OR = {
					has_ideology = ideology:ideology_anarchist_leader
					has_ideology = ideology:ideology_marxist_leader
					has_ideology = ideology:ideology_state_socialist_leader
					has_ideology = ideology:ideology_fascist_leader
					has_ideology = ideology:ideology_jihadist_leader
					has_ideology = ideology:ideology_theocrat_leader
					has_ideology = ideology:ideology_monarchist_leader
					has_ideology = ideology:ideology_right_wing_populist_leader
					has_ideology = ideology:ideology_authoritarian_capitalist_leader
					has_ideology = ideology:ideology_state_capitalist_leader
				}
			}
		}
		OR = {
			leader ?= { has_ideology = ideology:ideology_liberal_leader }
			leader ?= { has_ideology = ideology:ideology_social_liberal_leader }
			leader ?= { has_ideology = ideology:ideology_neoliberal_leader }
			leader ?= { has_ideology = ideology:ideology_progressive_leader }
			leader ?= { has_ideology = ideology:ideology_moderate_leader }
			leader ?= { has_ideology = ideology:ideology_technocrat_leader }
			leader ?= { has_ideology = ideology:ideology_classical_liberal_leader }
			leader ?= { has_ideology = ideology:ideology_democratic_socialist_leader }
			leader ?= { has_ideology = ideology:ideology_green_leader }
			leader ?= { has_ideology = ideology:ideology_left_wing_populist_leader }
			has_ideology = ideology:ideology_centre_economy
			has_ideology = ideology:ideology_individualist
			has_ideology = ideology:ideology_republican
			has_ideology = ideology:ideology_globalist
			has_ideology = ideology:ideology_right_economy
			has_ideology = ideology:ideology_progressive
			has_ideology = ideology:ideology_regionalist
			has_ideology = ideology:ideology_left_economy
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_professionals
			is_interest_group_type = ig_trade_unions
		}
	}

	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}

		if = {
			limit = {
				owner = {
					exists = c:USA
					THIS = c:USA
				}
			}
			add = { value = 100 desc = "two_party_system" }
		}

		if = {
			limit = {
				OR = {
					leader ?= { has_ideology = ideology:ideology_liberal_leader }
					leader ?= { has_ideology = ideology:ideology_social_liberal_leader }
					leader ?= { has_ideology = ideology:ideology_neoliberal_leader }
				}
			}
			add = { value = 120 desc = "leader_ideological_compatibility" }
		}
		if = {
			limit = {
				OR = {
					leader ?= { has_ideology = ideology:ideology_progressive_leader }
					leader ?= { has_ideology = ideology:ideology_moderate_leader }
					leader ?= { has_ideology = ideology:ideology_technocrat_leader }
					leader ?= { has_ideology = ideology:ideology_classical_liberal_leader }
					leader ?= { has_ideology = ideology:ideology_democratic_socialist_leader }
					leader ?= { has_ideology = ideology:ideology_green_leader }
					leader ?= { has_ideology = ideology:ideology_left_wing_populist_leader }
				}
			}
			add = { value = 40 desc = "leader_adjacent_ideology" }
		}
		if = {
			limit = {
				leader ?= {
					NOR = {
						has_ideology = ideology:ideology_liberal_leader
						has_ideology = ideology:ideology_social_liberal_leader
						has_ideology = ideology:ideology_neoliberal_leader
						has_ideology = ideology:ideology_progressive_leader
						has_ideology = ideology:ideology_moderate_leader
						has_ideology = ideology:ideology_technocrat_leader
						has_ideology = ideology:ideology_classical_liberal_leader
						has_ideology = ideology:ideology_democratic_socialist_leader
						has_ideology = ideology:ideology_green_leader
						has_ideology = ideology:ideology_left_wing_populist_leader
					}
				}
			}
			add = { value = -40 desc = "leader_ideology_unrelated" }
		}
		if = {
			limit = {
				leader ?= { has_ideology = ideology:ideology_moderate_leader }
				year < 2000
			}
			add = { value = 180 desc = "new_democrats_era" }
		}
		if = {
			limit = {
				OR = {
					leader ?= { has_ideology = ideology:ideology_marxist_leader }
					leader ?= { has_ideology = ideology:ideology_state_socialist_leader }
					leader ?= { has_ideology = ideology:ideology_fascist_leader }
					leader ?= { has_ideology = ideology:ideology_jihadist_leader }
					leader ?= { has_ideology = ideology:ideology_theocrat_leader }
					leader ?= { has_ideology = ideology:ideology_monarchist_leader }
				}
			}
			add = { value = -150 desc = "leader_ideological_incompatibility" }
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_centre_economy
					has_ideology = ideology:ideology_individualist
					has_ideology = ideology:ideology_republican
					has_ideology = ideology:ideology_globalist
				}
			}
			add = { value = 20 desc = "interest_group_ideological_compatibility" }
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_right_economy
					has_ideology = ideology:ideology_progressive
					has_ideology = ideology:ideology_regionalist
					has_ideology = ideology:ideology_left_economy
				}
			}
			add = { value = 8 desc = "interest_group_adjacent_ideology" }
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_far_left_economy
					has_ideology = ideology:ideology_totalitarian
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_theocrat
					has_ideology = ideology:ideology_monarchist
				}
			}
			add = { value = -20 desc = "interest_group_ideological_incompatibility" }
		}
		if = {
			limit = {
				OR = {
					is_interest_group_type = ig_intelligentsia
					is_interest_group_type = ig_professionals
					is_interest_group_type = ig_trade_unions
				}
			}
			add = { value = 15 desc = "from_archetype_party" }
		}
		if = {
			limit = {
				owner = { OR = { has_law = law_type:law_presidential_republic has_law = law_type:law_parliamentary_republic } }
			}
			add = { value = 10 desc = "democratic_government" }
		}
		if = {
			limit = {
				is_powerful = yes
				OR = {
					leader ?= { has_ideology = ideology:ideology_liberal_leader }
					leader ?= { has_ideology = ideology:ideology_social_liberal_leader }
					leader ?= { has_ideology = ideology:ideology_neoliberal_leader }
					has_ideology = ideology:ideology_centre_economy
					has_ideology = ideology:ideology_individualist
					has_ideology = ideology:ideology_republican
					has_ideology = ideology:ideology_globalist
				}
			}
			add = { value = 20 desc = "powerful_ig" }
		}
		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				is_marginal = no
			}
			multiply = { value = 2 desc = "from_influential_supporting_member" }
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = { value = 0.75 desc = "from_two_prior_members" }
		}
	}
}
